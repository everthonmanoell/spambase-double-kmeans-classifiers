def _markdown_table(frame):
    """Format a pandas DataFrame as a GitHub-flavored Markdown table.

    Args:
        frame (pd.DataFrame): Input table to format.

    Returns:
        str: Markdown-formatted table string.
    """
    columns = [str(column) for column in frame.columns]
    lines = ["| " + " | ".join(columns) + " |",
             "| " + " | ".join("---" for _ in columns) + " |"]
    for row in frame.itertuples(index=False, name=None):
        values = [f"{value:.8f}" if isinstance(value, float) else str(value)
                  for value in row]
        lines.append("| " + " | ".join(values) + " |")
    return "\n".join(lines)


def _ari_comment(ari):
    """Return an interpretive summary sentence for an Adjusted Rand Index value.

    Args:
        ari (float): Adjusted Rand Index score.

    Returns:
        str: Qualitative interpretation of ``ari``.
    """

            "particao spam/nao spam."
        )
    if np.isclose(ari, 0, rtol=0, atol=1e-12):
        return "A concordancia ajustada esta no nivel esperado ao acaso pelo ARI."
    return (
        "Ha concordancia positiva acima da referencia ao acaso, mas as particoes "
        "nao coincidem. O valor deve ser interpretado junto da tabela de contingencia."
    )


def _validate_winner(best, X):
    """Validate shapes, monotonicity, and reconstruction objective of a Q1 winner.

    Args:
        best (dict): Winning Double k-means run dictionary.
        X (np.ndarray): Standardized feature matrix of shape (n_samples, n_features).

    Raises:
        ValueError: If prototypes, column labels, or objective history are invalid.
    """
    K, H = best["K"], best["H"]
    columns = np.asarray(best["column_labels"])
    G = np.asarray(best["prototypes"])
    values = np.asarray(best["objective_history"])
    if (columns.shape != (X.shape[1],)
            or not np.issubdtype(columns.dtype, np.integer)
            or not np.array_equal(np.unique(columns), np.arange(H))
            or G.shape != (K, H) or not np.isfinite(G).all()
            or values.shape != (best["n_iter"] + 1,)
            or not np.isfinite(values).all()
            or not np.isclose(values[-1], best["objective"], rtol=1e-12)):
        raise ValueError("Invalid winning prototypes, column labels or objective history.")
    if np.any(np.diff(values) > 1e-12 * max(1.0, abs(values[0]))):
        raise ValueError("Winning objective history is not non-increasing.")
    reconstructed = G[np.ix_(best["labels"], columns)]
    if not np.isclose(np.sum((X - reconstructed) ** 2), best["objective"], rtol=1e-12):
        raise ValueError("Winning objective does not match the saved partition.")

def evaluate_clustering(filepath="dataset/spambase.data", results_dir="results",
                        plots_dir="plots", report_path="analysis/Q1_ANALYSIS.md"):
    """Regenerate silhouettes, evaluate their winner, and save all Q1 analysis.

    Args:
        filepath (str or Path, optional): Path to the Spambase dataset file.
            Defaults to "dataset/spambase.data".
        results_dir (str or Path, optional): Directory containing Q1 checkpoints
            and where evaluation outputs are saved. Defaults to "results".
        plots_dir (str or Path, optional): Directory where Q1 PNG plots are
            saved. Defaults to "plots".
        report_path (str or Path, optional): Path where the Q1 Markdown report
            is written. Defaults to "analysis/Q1_ANALYSIS.md".

    Returns:
        dict: Dictionary of summary metrics for the winning partition.

    Raises:
        ValueError: If the winning partition or original class vector is invalid.
    """
    results_dir, plots_dir = Path(results_dir), Path(plots_dir)
    scores = analyze_silhouette(filepath, results_dir, plots_dir)
    X, history = _load_analysis_inputs(Path(filepath), results_dir / "q1_full_history.pkl")
    selection = scores.loc[scores["silhouette"].idxmax()]
    runs = [run for run in history["runs"]
            if (run["K"], run["H"]) == (int(selection.K), int(selection.H))]
    best = min(runs, key=lambda run: run["objective"])
    _validate_winner(best, X)
    _, y = load_spambase_original(filepath)
    if not np.isin(y, [0, 1]).all() or len(np.unique(y)) != 2:
        raise ValueError("Spambase original classes must contain both 0 and 1.")

    labels = best["labels"]
    classes = np.unique(y).astype(int)
    contingency = pd.DataFrame(
        contingency_matrix(y, labels), index=classes, columns=np.arange(best["K"])
    ).rename_axis(index="original_class", columns="cluster")
    counts = np.bincount(labels, minlength=best["K"])
    sizes = pd.DataFrame({
        "cluster": np.arange(best["K"]), "n_objects": counts,
        "fraction": counts / len(y),
    })
    column_sizes = pd.DataFrame({
        "variable_cluster": np.arange(best["H"]),
        "n_features": np.bincount(best["column_labels"], minlength=best["H"]),
    })
    G = pd.DataFrame(
        best["prototypes"], columns=[f"variable_cluster_{h}" for h in range(best["H"])]
    ).rename_axis("object_cluster")
    objective = pd.DataFrame({
        "iteration": np.arange(len(best["objective_history"])),
        "objective": best["objective_history"],
    })
    metrics = {
        "K": best["K"], "H": best["H"], "best_run_id": best["run"],
        "silhouette": float(selection.silhouette),
        "best_objective": best["objective"], "best_converged": bool(best["converged"]),
        "n_iterations": best["n_iter"],
        "initial_objective": float(best["objective_history"][0]),
        "ari": float(adjusted_rand_score(y, labels)),
        "n_samples": X.shape[0], "n_features": X.shape[1],
        "n_runs_per_pair": history["metadata"]["n_runs"],
        "max_iter": history["metadata"]["max_iter"],
        "total_runs": len(history["runs"]),
        "converged_runs": sum(bool(run["converged"]) for run in history["runs"]),
        "largest_cluster_fraction": float(counts.max() / len(y)),
        "dataset_sha256": history["metadata"]["dataset_sha256"],
    }

    plot_q1_results(best, contingency, plots_dir)
    frames = {
        "q1_contingency.csv": contingency.reset_index(),
        "q1_cluster_sizes.csv": sizes,
        "q1_variable_cluster_sizes.csv": column_sizes,
        "q1_prototypes.csv": G.reset_index(),
        "q1_objective_history.csv": objective,
        "q1_object_labels.csv": pd.DataFrame({
            "row_index": np.arange(len(y)), "original_class": y.astype(int), "cluster": labels,
        }),
        "q1_variable_labels.csv": pd.DataFrame({
            "feature_index": np.arange(X.shape[1]), "cluster": best["column_labels"],
        }),
    }
    for name, frame in frames.items():
        with _atomic_output(results_dir / name) as stream:
            stream.write(frame.to_csv(index=False).encode("utf-8"))
    with _atomic_output(results_dir / "q1_selected_partition.npz") as stream:
        np.savez_compressed(
            stream, **best, scaler_mean=history["scaler_mean"],
            scaler_scale=history["scaler_scale"],
            dataset_sha256=history["metadata"]["dataset_sha256"],
        )
    with _atomic_output(results_dir / "q1_evaluation.json") as stream:
        stream.write(json.dumps(metrics, indent=2, allow_nan=False).encode("utf-8"))
    with _atomic_output(results_dir / "ari_score.txt") as stream:
        stream.write(f"ARI = {metrics['ari']:.10f}\n{_ari_comment(metrics['ari'])}\n".encode())
    _write_report(Path(report_path), metrics, scores, contingency, sizes,
                  column_sizes, G, plots_dir, results_dir)
    LOGGER.info("Q1 evaluated: K=%d H=%d ARI=%.10f", best["K"], best["H"], metrics["ari"])
    return metrics


def main(argv=None):
    """Parse CLI arguments and run the Question 1 clustering evaluation.

    Args:
        argv (list[str] or None, optional): Command-line argument list. Defaults
            to None.
    """
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--data", default="dataset/spambase.data")
    parser.add_argument("--results-dir", default="results")
    parser.add_argument("--plots-dir", default="plots")
    parser.add_argument("--report-path", default="analysis/Q1_ANALYSIS.md")
    args = parser.parse_args(argv)
    logging.basicConfig(level=logging.INFO, format="%(levelname)s: %(message)s")
    metrics = evaluate_clustering(args.data, args.results_dir, args.plots_dir, args.report_path)
    print(f"K*={metrics['K']}, H*={metrics['H']}, ARI={metrics['ari']:.10f}")
    print(f"Report: {args.report_path}")


if __name__ == "__main__":
    main()

"""Compare the nine Q1 winners by full-data Euclidean silhouette."""

import argparse
import hashlib
import json
import logging
from pathlib import Path
import pickle

from matplotlib.figure import Figure
import numpy as np
import pandas as pd
import sklearn
from sklearn.metrics import silhouette_score

from src.experiments.question_1 import Q1_GRID, _atomic_output
from src.utils.dataloader import load_spambase_original


LOGGER = logging.getLogger(__name__)


def _load_analysis_inputs(filepath, history_path):
    """Validate Q1 checkpoint integrity and return standardized features.

    Args:
        filepath (Path): Path to the raw Spambase dataset file.
        history_path (Path): Path to the saved ``q1_full_history.pkl`` file.

    Returns:
        tuple[np.ndarray, dict]: Standardized feature matrix ``X`` and the
            validated Q1 history dictionary.

    Raises:
        ValueError: If checkpoint metadata, versions, SHA-256 hash, run order,
            or standardization parameters are invalid.
    """
    # Pickle must only be loaded from a trusted local experiment.
    with history_path.open("rb") as stream:
        history = pickle.load(stream)
    metadata = history["metadata"]
    if (metadata["format_version"] != 1
            or metadata["preprocessing"] != "StandardScaler"
            or metadata["grid"] != Q1_GRID):
        raise ValueError("Unsupported Q1 checkpoint format, preprocessing or grid.")
    if hashlib.sha256(filepath.read_bytes()).hexdigest() != metadata["dataset_sha256"]:
        raise ValueError("Dataset differs from the Q1 checkpoint (SHA-256 mismatch).")
    for name, version in (
        ("numpy_version", np.__version__),
        ("pandas_version", pd.__version__),
        ("sklearn_version", sklearn.__version__),
    ):
        if metadata[name] != version:
            raise ValueError(f"{name} differs from the Q1 checkpoint.")

    n_runs = metadata["n_runs"]
    if not isinstance(n_runs, int) or isinstance(n_runs, bool) or n_runs < 1:
        raise ValueError("Checkpoint n_runs must be a positive integer.")
    expected = [(K, H, run) for K, H in Q1_GRID for run in range(n_runs)]
    actual = [(run["K"], run["H"], run["run"]) for run in history["runs"]]
    if actual != expected:
        raise ValueError("Analysis requires a complete Q1 history for all nine pairs.")
    if not all(np.isfinite(run["objective"]) for run in history["runs"]):
        raise ValueError("Q1 history contains non-finite objectives.")

    X, _ = load_spambase_original(filepath)
    mean = np.asarray(history["scaler_mean"])
    scale = np.asarray(history["scaler_scale"])
    if (X.shape != metadata["shape"]
            or mean.shape != (X.shape[1],) or scale.shape != mean.shape
            or not np.isfinite(X).all() or not np.isfinite(mean).all()
            or not np.isfinite(scale).all() or np.any(scale <= 0)):
        raise ValueError("Invalid feature matrix or saved standardization parameters.")
    # StandardScaler.transform performs these two operations, in this order.
    X = X.astype(float, copy=True)
    X -= mean
    X /= scale
    return X, history


def _heatmap(scores):
    """Build a Matplotlib heatmap figure for the (K, H) silhouette matrix.

    Args:
        scores (pd.DataFrame): DataFrame containing ``K``, ``H``, and
            ``silhouette`` columns.

    Returns:
        Figure: Rendered Matplotlib figure instance.
    """
    matrix = scores.pivot(index="K", columns="H", values="silhouette").reindex(
        index=[2, 3, 4], columns=[1, 2, 3, 4]
    )
    figure = Figure(figsize=(7, 4.5), layout="constrained")
    axes = figure.subplots()
    image = axes.imshow(
        np.ma.masked_invalid(matrix.to_numpy()), cmap="viridis", vmin=-1, vmax=1
    )
    axes.set_xticks(range(4), labels=[1, 2, 3, 4])
    axes.set_yticks(range(3), labels=[2, 3, 4])
    axes.set_xlabel("H (variable clusters)")
    axes.set_ylabel("K (object clusters)")
    winner = scores.loc[scores["silhouette"].idxmax()]
    axes.set_title(f"Q1 silhouette: K*={int(winner.K)}, H*={int(winner.H)}")
    for row in scores.itertuples(index=False):
        axes.text(row.H - 1, row.K - 2, f"{row.silhouette:.4f}",
                  ha="center", va="center", color="white")
    figure.colorbar(image, ax=axes, label="Euclidean silhouette")
    return figure


def analyze_silhouette(filepath="dataset/spambase.data", results_dir="results",
                       plots_dir="plots"):
    """Score each minimum-objective Q1 winner, without refitting clustering.

    The checkpoint is authoritative. Original labels are not used; features
    retain their original row order and use the saved Q1 standardization.
    Exact silhouette ties are resolved by ascending K, then H.

    Args:
        filepath (str or Path, optional): Path to the Spambase dataset file.
            Defaults to "dataset/spambase.data".
        results_dir (str or Path, optional): Directory containing Q1 artifacts
            and where silhouette outputs are saved. Defaults to "results".
        plots_dir (str or Path, optional): Directory where the silhouette
            heatmap PNG is saved. Defaults to "plots".

    Returns:
        pd.DataFrame: Nine-row DataFrame with the selected run and silhouette
            score per (K, H) pair.

    Raises:
        ValueError: If winner cluster labels are invalid.
        RuntimeError: If a computed silhouette score is non-finite.
    """
    results_dir, plots_dir = Path(results_dir), Path(plots_dir)
    X, history = _load_analysis_inputs(
        Path(filepath), results_dir / "q1_full_history.pkl"
    )
    if history["metadata"]["n_runs"] != 100:
        LOGGER.warning(
            "Reduced experiment: %d starts per pair, not the full 100-start Q1.",
            history["metadata"]["n_runs"],
        )
    rows = []
    for K, H in Q1_GRID:
        runs = [run for run in history["runs"] if (run["K"], run["H"]) == (K, H)]
        best = min(runs, key=lambda run: run["objective"])
        labels = np.asarray(best["labels"])
        if (labels.shape != (X.shape[0],)
                or not np.issubdtype(labels.dtype, np.integer)
                or not np.array_equal(np.unique(labels), np.arange(K))):
            raise ValueError(f"Invalid winner labels for K={K}, H={H}.")
        if not best["converged"]:
            LOGGER.warning("Winner K=%d H=%d did not converge.", K, H)
        score = float(silhouette_score(X, labels, metric="euclidean"))
        if not np.isfinite(score):
            raise RuntimeError(f"Non-finite silhouette for K={K}, H={H}.")
        rows.append({
            "K": K,
            "H": H,
            "n_runs": len(runs),
            "best_run_id": best["run"],
            "best_objective": best["objective"],
            "best_converged": bool(best["converged"]),
            "silhouette": score,
        })
        LOGGER.info("K=%d H=%d seed=%d silhouette=%.8f", K, H, best["run"], score)

    scores = pd.DataFrame(rows)
    selection = scores.loc[scores["silhouette"].idxmax()].to_dict()
    selection.update({
        "metric": "euclidean",
        "preprocessing": "StandardScaler",
        "dataset_sha256": history["metadata"]["dataset_sha256"],
    })
    plots_dir.mkdir(parents=True, exist_ok=True)
    figure = _heatmap(scores)
    try:
        with _atomic_output(plots_dir / "silhouette_heatmap.png") as stream:
            figure.savefig(stream, format="png", dpi=160)
    finally:
        figure.clear()
    with _atomic_output(results_dir / "silhouette_scores.csv") as stream:
        stream.write(scores.to_csv(index=False).encode("utf-8"))
    with _atomic_output(results_dir / "q1_selection.json") as stream:
        stream.write(json.dumps(selection, indent=2, allow_nan=False).encode("utf-8"))
    return scores


def main(argv=None):
    """Parse CLI arguments and run the Q1 silhouette analysis.

    Args:
        argv (list[str] or None, optional): Command-line argument list. Defaults
            to None.
    """
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--data", default="dataset/spambase.data")
    parser.add_argument("--results-dir", default="results")
    parser.add_argument("--plots-dir", default="plots")
    args = parser.parse_args(argv)
    logging.basicConfig(level=logging.INFO, format="%(levelname)s: %(message)s")
    scores = analyze_silhouette(args.data, args.results_dir, args.plots_dir)
    print(scores.to_string(index=False))
    winner = scores.loc[scores["silhouette"].idxmax()]
    print(f"\nK* = {int(winner.K)}, H* = {int(winner.H)}, "
          f"silhouette = {winner.silhouette:.8f}")


if __name__ == "__main__":
    main()

EXPECTED_REPETITIONS = 30
EXPECTED_FOLDS_PER_REPETITION = 10


def validate_results(df_results: pd.DataFrame) -> None:
    """Validate that the DataFrame contains a complete, aligned 30×10 experiment.

    Args:
        df_results (pd.DataFrame): Cross-validation results DataFrame loaded from
            ``cv_30x10_results.csv``.

    Raises:
        ValueError: If required columns are missing, row counts or categories
            do not match the 30×10 protocol, values are non-numeric/non-finite
            or outside [0, 1], or folds are misaligned.
    """
    required_columns = {
        "fold_id",
        "repetition",
        "fold_in_rep",
        "dataset",
        "classifier",
        *METRICS,
    }
    missing_columns = required_columns.difference(df_results.columns)
    if missing_columns:
        raise ValueError(f"Results CSV is missing columns: {sorted(missing_columns)}")

    expected_rows = (
        EXPECTED_REPETITIONS
        * EXPECTED_FOLDS_PER_REPETITION
        * len(DATASET_NAMES)
        * len(CLASSIFIER_NAMES)
    )
    if len(df_results) != expected_rows:
        raise ValueError(f"Expected {expected_rows} result rows, found {len(df_results)}.")
    if set(df_results["dataset"].unique()) != set(DATASET_NAMES):
        raise ValueError("Results must contain exactly the original and kstar datasets.")
    if set(df_results["classifier"].unique()) != set(CLASSIFIER_NAMES):
        raise ValueError("Results must contain all five expected classifiers.")
    if df_results.duplicated(["dataset", "classifier", "fold_id"]).any():
        raise ValueError("Results contain duplicate dataset/classifier/fold rows.")

    for column in ("fold_id", "repetition", "fold_in_rep", *METRICS):
        try:
            values = pd.to_numeric(df_results[column], errors="raise").to_numpy(dtype=float)
        except (TypeError, ValueError) as error:
            raise ValueError(f"Column {column!r} must contain numeric values.") from error
        if not np.isfinite(values).all():
            raise ValueError(f"Column {column!r} contains non-finite values.")

    for metric in METRICS:
        if not df_results[metric].between(0, 1).all():
            raise ValueError(f"Metric {metric!r} must be between 0 and 1.")

    expected_keys = {
        (repetition, fold_in_rep, (repetition - 1) * EXPECTED_FOLDS_PER_REPETITION + fold_in_rep - 1)
        for repetition in range(1, EXPECTED_REPETITIONS + 1)
        for fold_in_rep in range(1, EXPECTED_FOLDS_PER_REPETITION + 1)
    }
    for dataset_name in DATASET_NAMES:
        for classifier_name in CLASSIFIER_NAMES:
            subset = df_results.loc[
                (df_results["dataset"] == dataset_name)
                & (df_results["classifier"] == classifier_name)
            ]
            actual_keys = set(
                subset[["repetition", "fold_in_rep", "fold_id"]]
                .itertuples(index=False, name=None)
            )
            if actual_keys != expected_keys:
                raise ValueError(
                    f"Incomplete or misaligned 30×10 folds for "
                    f"{dataset_name}/{classifier_name}."
                )


def compute_confidence_intervals(
    df_results: pd.DataFrame,
    confidence_level: float = 0.95,
) -> pd.DataFrame:
    """Compute t-based CIs from the 30 per-repetition mean scores.

    Averaging the ten folds within each repetition avoids treating all 300
    overlapping outer folds as independent observations for these intervals.

    Args:
        df_results (pd.DataFrame): Aligned 30×10 cross-validation results.
        confidence_level (float, optional): Confidence level in (0, 1).
            Defaults to 0.95.

    Returns:
        pd.DataFrame: Table containing ``dataset``, ``classifier``, ``metric``,
            ``mean``, ``ci_lower``, ``ci_upper``, ``n_repetitions``, and
            ``confidence_level``.

    Raises:
        ValueError: If ``confidence_level`` is outside (0, 1), required columns
            are missing, or fewer than two repetitions are present.
    """
    if not 0 < confidence_level < 1:
        raise ValueError("confidence_level must be between 0 and 1.")
    if "repetition" not in df_results or "dataset" not in df_results or "classifier" not in df_results:
        raise ValueError("Results must include repetition, dataset, and classifier columns.")
    missing_metrics = set(METRICS).difference(df_results.columns)
    if missing_metrics:
        raise ValueError(f"Results are missing metrics: {sorted(missing_metrics)}")

    rows = []
    for (dataset_name, classifier_name), group in df_results.groupby(
        ["dataset", "classifier"], sort=False
    ):
        for metric in METRICS:
            repetition_means = group.groupby("repetition", sort=True)[metric].mean()
            n_repetitions = len(repetition_means)
            if n_repetitions < 2:
                raise ValueError("At least two repetitions are required to compute a CI.")
            mean = float(repetition_means.mean())
            margin = float(
                t.ppf((1 + confidence_level) / 2, df=n_repetitions - 1)
                * sem(repetition_means.to_numpy())
            )
            rows.append(
                {
                    "dataset": dataset_name,
                    "classifier": classifier_name,
                    "metric": metric,
                    "mean": mean,
                    "ci_lower": mean - margin,
                    "ci_upper": mean + margin,
                    "n_repetitions": n_repetitions,
                    "confidence_level": confidence_level,
                }
            )
    return pd.DataFrame(rows)


def _print_confidence_intervals(confidence_intervals: pd.DataFrame) -> None:
    """Print formatted point estimates and 95% confidence intervals to stdout.

    Args:
        confidence_intervals (pd.DataFrame): Output of
            ``compute_confidence_intervals``.
    """
    print("=" * 96)
    print("MÉTRICAS E IC 95% — MÉDIA DAS 30 REPETIÇÕES (10 FOLDS POR REPETIÇÃO)")
    print("=" * 96)
    for dataset_name in DATASET_NAMES:
        print(f"\nDataset: {dataset_name}")
        subset = confidence_intervals[confidence_intervals["dataset"] == dataset_name]
        for classifier_name in CLASSIFIER_NAMES:
            classifier = subset[subset["classifier"] == classifier_name]
            metrics = " | ".join(
                f"{METRIC_LABELS[row.metric]}={row.mean:.4f} "
                f"[{row.ci_lower:.4f}, {row.ci_upper:.4f}]"
                for row in classifier.itertuples(index=False)
            )
            print(f"  {classifier_name:20s} | {metrics}")


def _plot_metric_distributions(df_results: pd.DataFrame, output_path: Path) -> None:
    """Generate and save boxplots comparing classifiers across datasets and metrics.

    Args:
        df_results (pd.DataFrame): Aligned 30×10 cross-validation results.
        output_path (Path): Destination file path for the PNG plot.

        nrows=len(METRICS),
        ncols=len(DATASET_NAMES),

        "Bayesian\nk-NN",
        "Parzen",

                dataset.loc[dataset["classifier"] == classifier_name, metric].to_numpy()

                    "marker": "D",
                    "markerfacecolor": "white",
                    "markeredgecolor": "black",
                    "markersize": 4,
                },
            )
            for box, color in zip(boxes["boxes"], colors):
                box.set_facecolor(color)
                box.set_alpha(0.75)
            if row_index == 0:
                axis.set_title(
                    dataset_name.replace("_", " "),
                    fontsize=12,
                    fontweight="bold",
                    pad=12,
                )
            if column_index == 0:
                axis.set_ylabel(METRIC_LABELS[metric], fontsize=10)

            axis.set_xticklabels(classifier_labels, fontsize=9)
            axis.yaxis.set_major_formatter(plt.FuncFormatter(lambda value, _: f"{value:.3f}"))
            all_values = np.concatenate(values)
            lower, upper = np.quantile(all_values, [0.02, 0.98])
            padding = max((upper - lower) * 0.12, np.finfo(float).eps)
            axis.set_ylim(lower - padding, upper + padding)

    figure.suptitle(

        fontsize=15,
    )
    figure.tight_layout(rect=(0, 0, 1, 0.955), h_pad=2.2, w_pad=1.8)
    figure.savefig(output_path, dpi=240, bbox_inches="tight")
    plt.close(figure)


def run_statistical_analysis(
    input_csv: str | Path = "results/q2/cv_30x10_results.csv",
    results_dir: str | Path = "results",
    plots_dir: str | Path = "plots",
    alpha: float = 0.05,
) -> pd.DataFrame:
    """Analyze the 30×10 CSV and save/print the requested 2.3 outputs.

    Args:
        input_csv (str or Path, optional): Path to the 30×10 CV results CSV.
            Defaults to "results/q2/cv_30x10_results.csv".
        results_dir (str or Path, optional): Directory where CSV and TXT
            statistical summaries are saved. Defaults to "results".
        plots_dir (str or Path, optional): Directory where the comparison
            boxplot PNG is saved. Defaults to "plots".
        alpha (float, optional): Significance threshold for Friedman and
            Nemenyi tests. Defaults to 0.05.

    Returns:
        pd.DataFrame: DataFrame of point estimates and confidence intervals.

    Raises:
        ValueError: If the input CSV fails ``validate_results``.
    """
    input_csv = Path(input_csv)
    results_dir = Path(results_dir)
    plots_dir = Path(plots_dir)
    df_results = pd.read_csv(input_csv)
    validate_results(df_results)
    results_dir.mkdir(parents=True, exist_ok=True)
    plots_dir.mkdir(parents=True, exist_ok=True)

    confidence_intervals = compute_confidence_intervals(df_results)
    confidence_path = results_dir / "confidence_intervals.csv"
    confidence_intervals.to_csv(confidence_path, index=False)
    _print_confidence_intervals(confidence_intervals)
    print(f"\nIntervalos de confiança salvos em: {confidence_path}")

    test_summaries = []
    nemenyi_matrices = []
    for dataset_name in DATASET_NAMES:
        dataset = df_results[df_results["dataset"] == dataset_name]
        for metric in METRICS:
            aligned = dataset.pivot(
                index="fold_id",
                columns="classifier",
                values=metric,
            ).reindex(columns=CLASSIFIER_NAMES).sort_index()
            result = perform_friedman_nemenyi(
                aligned,
                metric_name=f"{dataset_name} — {METRIC_LABELS[metric]}",
                alpha=alpha,
            )
            test_summaries.append(
                {
                    "dataset": dataset_name,
                    "metric": metric,
                    "friedman_statistic": result.statistic,
                    "p_value": result.p_value,
                    "alpha": alpha,
                    "significant": result.p_value < alpha,
                    "nemenyi_performed": result.nemenyi_p_values is not None,
                }
            )
            if result.nemenyi_p_values is not None:
                matrix = result.nemenyi_p_values.reindex(
                    index=CLASSIFIER_NAMES,
                    columns=CLASSIFIER_NAMES,
                )
                matrix.index.name = "classifier"
                matrix = matrix.reset_index()
                matrix.insert(0, "metric", metric)
                matrix.insert(0, "dataset", dataset_name)
                nemenyi_matrices.append(matrix)

    tests_path = results_dir / "friedman_results.txt"
    with tests_path.open("w", encoding="utf-8") as output:
        output.write("TESTES DE FRIEDMAN E PÓS-TESTE DE NEMENYI\n")
        output.write(f"Alpha: {alpha}\n")
        output.write("Bloco: fold externo alinhado entre classificadores (300 por dataset).\n\n")
        for result in test_summaries:
            output.write(
                f"{result['dataset']} | {METRIC_LABELS[result['metric']]}: "
                f"chi-square={result['friedman_statistic']:.6f}, "
                f"p={result['p_value']:.6e}, "
                f"significativo={result['significant']}, "
                f"Nemenyi executado={result['nemenyi_performed']}\n"
            )
        output.write(
            "\nNota: folds e repetições se sobrepõem; os p-valores não devem ser "
            "interpretados como evidência de generalização para outros datasets.\n"
        )
    print(f"\nResultados de Friedman salvos em: {tests_path}")

    classifier_columns = ["dataset", "metric", "classifier", *CLASSIFIER_NAMES]
    nemenyi_path = results_dir / "nemenyi_pvalues.csv"
    nemenyi_output = (
        pd.concat(nemenyi_matrices, ignore_index=True)
        if nemenyi_matrices
        else pd.DataFrame(columns=classifier_columns)
    )
    nemenyi_output.to_csv(nemenyi_path, index=False)
    print(f"Matrizes de Nemenyi salvas em: {nemenyi_path}")

    plot_path = plots_dir / "comparison_boxplots.png"
    _plot_metric_distributions(df_results, plot_path)
    print(f"Gráfico comparativo salvo em: {plot_path}")
    return confidence_intervals

## Decisões de Arquitetura: Classificador Bayesiano Baseado em k-Vizinhos

### 1. Justificativa Teórica: A Equivalência Bayesiana
O projeto exige a implementação de um classificador bayesiano utilizando a estimativa de densidade por k-vizinhos e a estimativa de máxima verossimilhança para a probabilidade a priori das classes $P(\omega_i)$. Matematicamente, a regra de classificação padrão do algoritmo k-NN é idêntica à formulação bayesiana.

Na estimativa de densidade baseada em vizinhança, a função de verossimilhança é dada por:
$$p(x\vert{} \omega_i) \approx \frac{k_i}{N_i V}$$
Onde $k_i$ é o número de vizinhos da classe $\omega_i$ dentro da vizinhança de volume $V$, e $N_i$ é o total de exemplos da classe. O volume $V$ é o mesmo para todas as classes, pois a vizinhança é construída globalmente em torno de $x$, contendo $k$ pontos de todas as classes. Isso garante que, independentemente da métrica (Euclidiana, City-Block ou Chebyshev), os termos $V$, $N$ e $N_i$ se cancelem. Utilizando a estimativa de máxima verossimilhança para a priori ($P(\omega_i) = \frac{N_i}{N}$) e aplicando o Teorema de Bayes:
$$P(\omega_i \vert{} x) = \frac{p(x \vert{} \omega_i) P(\omega_i)}{p(x)} = \frac{\frac{k_i}{N_i V} \frac{N_i}{N}}{\frac{k}{N V}} = \frac{k_i}{k}$$
Como resultado, alocar a amostra à classe de maior probabilidade a posteriori equivale a escolher a classe majoritária entre os $k$ vizinhos. Por isso, a classe nativa do Scikit-Learn foi envelopada: com pesos uniformes, seu `predict_proba` já devolve exatamente a razão $\frac{k_i}{k}$.

### 2. Pré-processamento
As distâncias são calculadas sobre os atributos padronizados (`StandardScaler`), de modo que $V$ é medido no espaço padronizado e nenhum atributo domina a métrica por escala. O `StandardScaler` e o `KNeighborsClassifier` formam um `Pipeline` interno, ajustado dentro do `fit`. Assim, em cada fold da validação cruzada, a média e o desvio-padrão vêm apenas do conjunto de treino, sem vazamento de informação do conjunto de teste.

### 3. Definição da Grade de Busca (Hyperparameter Grid)
- **Métrica de Distância (`metric`):** Em atendimento ao documento do projeto, a grade testa as distâncias Euclidiana, City-Block (`cityblock` no Scikit-Learn, também chamada de Manhattan) e Chebyshev.
- **Número de Vizinhos (`n_neighbors`):** Optou-se por uma grade estendida de valores ímpares (`[1, 3, 5, 7, 11, 15, 21, 31, 45, 63, 91]`). É importante notar que $k$ ímpar previne empates apenas em decisões binárias. Como a segunda versão do dataset terá $K^*$ classes obtidas por agrupamento (ex: 3 ou 4), empates na contagem de votos podem ocorrer (ex: $k=3$ com votos 1-1-1, ou $k=5$ com 2-2-1). Nesses casos, a implementação do Scikit-Learn desempata escolhendo a classe de menor rótulo numérico (comportamento herdado do `np.argmax` sobre as classes ordenadas), e não pela distância ao vizinho mais próximo. A progressão não linear permite ao `GridSearchCV` explorar desde contornos ruidosos ($k=1$) até valores acima da ordem de grandeza $\sqrt{N}$ do conjunto de treino ($\approx 58$ a $64$ no Spambase). O limite superior ($k=91$) fica deliberadamente além desse valor, para verificar se uma suavização ainda maior melhora a validação interna.


## Decisões de Arquitetura e Hiperparâmetros: Regressão Logística

### 1. Arquitetura e Encapsulamento (Wrapper)
* **Padronização de Interface:** A classe `CustomLogisticRegression` foi construída como um *wrapper* (encapsulamento) herdando de `BaseEstimator` e `ClassifierMixin`. Essa decisão garante que o modelo nativo do Scikit-Learn tenha exatamente a mesma assinatura e comportamento das classes construídas do zero (como a Janela de Parzen). Isso mantém o script principal do experimento limpo e modular.
* **Isolamento da Grade de Busca:** A implementação do método `@staticmethod get_param_grid()` acopla o dicionário de hiperparâmetros diretamente à classe do modelo. Isso evita a poluição do script principal com dicionários soltos e garante que a configuração de busca pertença exclusivamente ao classificador correspondente.

### 2. Tratamento de Convergência e Escala
* **Ajuste de Iterações (`max_iter=10000`):** O conjunto de dados *Spambase* possui 57 atributos contínuos em escalas numéricas muito discrepantes (frequências percentuais versus contagens de comprimento de sequências de letras maiúsculas). O limite padrão de 100 iterações do Scikit-Learn é insuficiente para o algoritmo convergir para o mínimo global nesse cenário. O aumento para 10000 iterações como padrão melhora a possibilidade de convergência matemática correta e evita interrupções por avisos (*warnings*) de limite excedido durante o treinamento.

### 3. Definição da Grade de Busca (Hyperparameter Grid)
* **Força de Regularização (`C`):** A grade escolhida foi `[0.01, 0.1, 1.0, 10.0, 100.0]`. Utilizou-se uma escala logarítmica para explorar diferentes ordens de grandeza da regularização inversa. O espaço de busca cobre desde um modelo altamente penalizado e protegido contra *overfitting* (`0.01`) até um modelo com flexibilidade quase total para se ajustar aos dados de treinamento (`100.0`).
* **Penalidade (`penalty='l2'`) e Otimizador (`solver='lbfgs'`):** A penalidade L2 (Ridge) foi fixada por ser a mais robusta e numericamente estável para problemas com múltiplas variáveis contínuas. O otimizador `lbfgs` foi escolhido por ser o algoritmo padrão e mais eficiente em termos de memória para funções com penalidade L2. 

### 4. Dimensionamento do Custo Computacional
- **Viabilidade da Validação Cruzada Aninhada:** A metodologia do projeto exige uma validação cruzada externa de 30 repetições de 10-folds, com um loop interno de 5-folds nos dados de treino para o ajuste dos hiperparâmetros.
- **Controle de Explosão Combinatória:** Essa estrutura aninhada gera uma carga computacional massiva. Ao fixar o otimizador e o tipo de penalidade, limitando a grade a 5 valores do parâmetro `C`, o total de treinamentos da Regressão Logística foi contido em 7.800 execuções (300 folds externos $\times$ (5 folds internos $\times$ 5 valores de C + 1 refit final)). Como o enunciado pede apenas a validação cruzada dos hiperparâmetros, optou-se por fixar a penalidade L2 e concentrar o ajuste exclusivamente no hiperparâmetro `C`.

## Decisões de Arquitetura: Classificador Bayesiano Gaussiano

* **Integração com o Ecossistema Scikit-Learn:** A classe `GaussianBayesClassifier` herda de `ClassifierMixin` e `BaseEstimator`, nessa ordem. Ela pode ser usada na validação cruzada aninhada da Questão 2a ($30 \times 10$ folds externos e 5 folds internos). O modelo não tem hiperparâmetros: `get_param_grid()` retorna `{}`.
* **Estimativas de Máxima Verossimilhança:** A priori $P(\omega_i)=N_i/N$, a média $\hat{\mu}_i$ e a covariância $\hat{\Sigma}_i$ são estimadas por MV. A covariância é calculada como $(X_i-\hat{\mu}_i)^\top(X_i-\hat{\mu}_i)/N_i$, sem correção não-viesada, e preservada em `covariances_`.
* **Estabilização Numérica para Covariância Singular (`reg_covar=1e-6`):** Na base `kstar_classes` ($N_1=34 < d=57$) e em subamostras pequenas da curva de aprendizagem (inclusive em `original_2_classes` com 5% de treino, onde variáveis raras têm variância zero), a covariância de MV é singular ($\det(\hat{\Sigma}_i)=0$). Para garantir que a matriz seja definida positiva na fatoração de Cholesky sem introduzir hiperparâmetros de busca, soma-se um termo fixo de estabilização numérica diagonal $\varepsilon I$ (`reg_covar=1e-6`, análogo ao padrão do `GaussianMixture` no Scikit-Learn). Caso `reg_covar=0.0` seja informado explicitamente, matrizes singulares lançam `ValueError`.
* **Cálculo via Decomposição de Cholesky:** O método `fit` armazena $L_i$, com $L_iL_i^\top=\hat{\Sigma}_i + \varepsilon I$. A predição resolve $L_i z=x-\hat{\mu}_i$ para obter a distância de Mahalanobis e calcula $\log|\hat{\Sigma}_i + \varepsilon I|=2\sum_j\log (L_i)_{jj}$, sem inversão explícita.
* **Regra de Decisão no Espaço Logarítmico:** Como no Parzen, o `.predict()` escolhe $\arg\max_i \left[\log p(x|\omega_i) + \log P(\omega_i)\right]$, já que o denominador da regra de Bayes é comum a todas as classes. O `.predict_proba()` retorna as posteriores $P(\omega_i|x)$ normalizadas com `logsumexp`, o que evita *underflow* das densidades em 57 dimensões. O item v do enunciado pede voto majoritário, que usa os rótulos previstos pelo `.predict()` de cada classificador (voto *hard*), e não essas probabilidades.


## Decisões de Arquitetura: Classificador Bayesiano Baseado em k-Vizinhos

### 1. Justificativa Teórica: A Equivalência Bayesiana
O projeto exige a implementação de um classificador bayesiano utilizando a estimativa de densidade por k-vizinhos e a estimativa de máxima verossimilhança para a probabilidade a priori das classes $P(\omega_i)$. Matematicamente, a regra de classificação padrão do algoritmo k-NN é idêntica à formulação bayesiana. 

Na estimativa de densidade baseada em vizinhança, a função de verossimilhança é dada por:
$$p(x \vert{} \omega_i) \approx \frac{k_i}{N_i V}$$
Onde $k_i$ é o número de vizinhos da classe $\omega_i$ dentro da vizinhança de volume $V$, e $N_i$ é o total de exemplos da classe. Utilizando a estimativa de máxima verossimilhança para a priori ($P(\omega_i) = \frac{N_i}{N}$), e aplicando o Teorema de Bayes para a probabilidade a posteriori:
$$P(\omega_i \vert{} x) = \frac{p(x \vert{} \omega_i) P(\omega_i)}{p(x)} = \frac{\frac{k_i}{N_i V} \frac{N_i}{N}}{\frac{k}{N V}} = \frac{k_i}{k}$$
Como resultado, a alocação da amostra à classe com a maior densidade a posteriori equivale diretamente a escolher a classe majoritária entre os $k$ vizinhos. Por esta razão, a classe nativa do Scikit-Learn foi envelopada, pois ela já processa eficientemente a razão $\frac{k_i}{k}$.

### 2. Definição da Grade de Busca (Hyperparameter Grid)
* **Métrica de Distância (`metric`):** Em estrito atendimento ao documento do projeto, a grade de busca fixará a vizinhança testando as distâncias Euclidiana, City-Block (conhecida como Manhattan na biblioteca) e Chebishev.
* **Número de Vizinhos (`n_neighbors`):** Optou-se por uma grade de valores ímpares (`[1, 3, 5, 7, 11, 15, 21, 31]`) para prevenir empates em deliberações majoritárias bidimensionais. A progressão não linear permite ao `GridSearchCV` investigar o comportamento do modelo desde contornos de decisão altamente ruidosos ($k=1$) até fronteiras mais suaves e globalizadas ($k=31$), sem inflar o custo da validação cruzada aninhada $30 \times 10$-folds.
## Decisões de Arquitetura: Classificador Janela de Parzen

* **Integração com o Ecossistema Scikit-Learn:** A classe customizada `ParzenWindowClassifier` herda de `BaseEstimator` e `ClassifierMixin`. Esta estrutura foi adotada para garantir compatibilidade nativa com o `GridSearchCV`, permitindo que o modelo seja injetado diretamente no pipeline de validação cruzada aninhada ($30 \times 10$-folds) exigido na Questão 2a, automatizando o ajuste da janela $h$ sem a necessidade de loops manuais de busca[cite: 4].
* **Mapeamento do Kernel Multivariado Produto:** O escopo do projeto demanda a aplicação de um kernel multivariado produto baseado no kernel gaussiano univariado[cite: 4]. Para isso, instanciou-se o `KernelDensity(kernel='gaussian')` separadamente para o subconjunto de dados de cada classe. Ao fornecer um parâmetro escalar único para a largura da janela (bandwidth $h$), a implementação realiza matematicamente o produto das densidades gaussianas univariadas independentes em todas as 57 dimensões, cumprindo o requisito de forma vetorizada e eficiente.
* **Estimativa de Máxima Verossimilhança das Priori:** O documento especifica o uso da estimativa de máxima verossimilhança para a obtenção das probabilidades a priori $P(\omega_i)$[cite: 4]. No método `.fit()`, isso foi rigorosamente implementado calculando a frequência relativa empírica de cada classe no conjunto de treinamento ($P(\omega_i) = \frac{N_i}{N_{total}}$).
* **Regra de Decisão no Espaço Logarítmico (Log-Space):** O método `.predict()` segue o Teorema de Bayes para alocar a amostra à classe com a maior probabilidade a posteriori. Para prevenir problemas críticos de *underflow* numérico — muito comuns ao multiplicar pequenas probabilidades de densidade em altas dimensões como a da base Spambase — a regra de decisão foi projetada para operar com somas no espaço logarítmico: $\arg\max_i (\log(p(x|\omega_i)) + \log(P(\omega_i)))$. Essa abordagem maximiza a estabilidade numérica extraindo diretamente a log-densidade através do método nativo `score_samples()`.

## Decisões de Arquitetura: Classificador Bayesiano Gaussiano

* **Integração com o Ecossistema Scikit-Learn:** A classe `GaussianBayesClassifier` herda de `ClassifierMixin` e `BaseEstimator`, nessa ordem. Ela pode ser usada na validação cruzada aninhada da Questão 2a ($30 \times 10$ folds externos e 5 folds internos). O modelo não tem hiperparâmetros: `get_param_grid()` retorna `{}`.
* **Estimativas de Máxima Verossimilhança:** A priori $P(\omega_i)=N_i/N$, a média $\hat{\mu}_i$ e a covariância $\hat{\Sigma}_i$ são estimadas por MV. A covariância é calculada como $(X_i-\hat{\mu}_i)^\top(X_i-\hat{\mu}_i)/N_i$, sem correção não-viesada, e preservada em `covariances_`.
* **Estabilização Numérica para Covariância Singular (`reg_covar=1e-6`):** Na base `kstar_classes` ($N_1=34 < d=57$) e em subamostras pequenas da curva de aprendizagem (inclusive em `original_2_classes` com 5% de treino, onde variáveis raras têm variância zero), a covariância de MV é singular ($\det(\hat{\Sigma}_i)=0$). Para garantir que a matriz seja definida positiva na fatoração de Cholesky sem introduzir hiperparâmetros de busca, soma-se um termo fixo de estabilização numérica diagonal $\varepsilon I$ (`reg_covar=1e-6`, análogo ao padrão do `GaussianMixture` no Scikit-Learn). Caso `reg_covar=0.0` seja informado explicitamente, matrizes singulares lançam `ValueError`.
* **Cálculo via Decomposição de Cholesky:** O método `fit` armazena $L_i$, com $L_iL_i^\top=\hat{\Sigma}_i + \varepsilon I$. A predição resolve $L_i z=x-\hat{\mu}_i$ para obter a distância de Mahalanobis e calcula $\log|\hat{\Sigma}_i + \varepsilon I|=2\sum_j\log (L_i)_{jj}$, sem inversão explícita.
* **Regra de Decisão no Espaço Logarítmico:** Como no Parzen, o `.predict()` escolhe $\arg\max_i \left[\log p(x|\omega_i) + \log P(\omega_i)\right]$, já que o denominador da regra de Bayes é comum a todas as classes. O `.predict_proba()` retorna as posteriores $P(\omega_i|x)$ normalizadas com `logsumexp`, o que evita *underflow* das densidades em 57 dimensões. O item v do enunciado pede voto majoritário, que usa os rótulos previstos pelo `.predict()` de cada classificador (voto *hard*), e não essas probabilidades.


## Decisões de Arquitetura: Classificador Baseado em Voto Majoritário

- **Estratégia de Votação (Hard Voting):** A classe `MajorityVotingClassifier` atende rigorosamente à exigência de utilizar a regra do voto majoritário a partir dos quatro classificadores base. A configuração `voting='hard'` foi definida no encapsulamento do Scikit-Learn para que a decisão do *ensemble* seja computada com base nas predições categóricas (rótulos finais) emitidas por cada modelo, em oposição à média das probabilidades a posteriori (*soft voting*). A classe designada à amostra é aquela que obtém a maioria simples dos votos.
- **Injeção de Dependência dos Classificadores:** O classificador de voto majoritário não instancia os algoritmos subjacentes em seu construtor. Em vez disso, recebe a lista estruturada de `estimators`. Esta abordagem arquitetural garante o isolamento da etapa de treinamento: a validação cruzada interna de 5-folds para ajuste de hiperparâmetros é executada individualmente no Bayesiano Gaussiano, K-Vizinhos, Janela de Parzen e Regressão Logística. Apenas os modelos já otimizados são posteriormente injetados no *ensemble* para compor a validação externa.
- **Resolução Determinística de Empates:** A exigência de combinar exatamente 4 classificadores estabelece um colégio eleitoral par, o que introduz o risco matemático de empates absolutos (ex: dois votos para a classe A, dois votos para a classe B). O uso da implementação nativa do Scikit-Learn garante um desempate determinístico, selecionando sistematicamente a classe com o menor valor numérico ou precedência lexicográfica, preservando a reprodutibilidade metodológica ao longo dos $30 \times 10$-folds.

import numpy as np
from scipy.linalg import solve_triangular
from scipy.special import logsumexp
from sklearn.base import BaseEstimator, ClassifierMixin
from sklearn.utils.validation import check_is_fitted, validate_data


class GaussianBayesClassifier(ClassifierMixin, BaseEstimator):
    """Bayesian classifier with multivariate normal class densities.

    Class priors, means, and covariance matrices use maximum-likelihood
    estimates (dividing by N_i). A small non-negative diagonal term
    ``reg_covar`` (default ``1e-6``) is added when computing the Cholesky
    factorization to guarantee positive definiteness when a class covariance
    is rank-deficient (e.g. N_i <= d or zero-variance features in small splits).

    Args:
        reg_covar (float, optional): Non-negative diagonal stabilization added
            to each class covariance matrix before Cholesky factorization.
            Defaults to 1e-6.
    """

    def __init__(self, reg_covar: float = 1e-6):
        self.reg_covar = reg_covar

    def fit(self, X, y):
        """Estimate class priors, means, covariances, and Cholesky factors.

        Args:
            X (array-like of shape (n_samples, n_features)): Training vectors.
            y (array-like of shape (n_samples,)): Target values.

        Returns:
            GaussianBayesClassifier: The fitted classifier.

        Raises:
            ValueError: If ``reg_covar`` is negative or if a class covariance
                matrix is singular or not positive definite.
        """
        if self.reg_covar < 0:
            raise ValueError(f"reg_covar must be non-negative, got {self.reg_covar}.")

        # Also stores n_features_in_ and rejects NaN/inf values.
        X, y = validate_data(self, X, y, dtype=float)

        self.classes_ = np.unique(y)
        n_classes = len(self.classes_)
        n_samples, n_features = X.shape

        self.priors_ = np.empty(n_classes)
        self.means_ = np.empty((n_classes, n_features))
        self.covariances_ = np.empty((n_classes, n_features, n_features))
        self.cholesky_factors_ = np.empty((n_classes, n_features, n_features))

        eye = np.eye(n_features)

        for i, c in enumerate(self.classes_):
            X_c = X[y == c]

            # ML estimates: P(w_i) = N_i / N; covariance divides by N_i (bias=True).
            self.priors_[i] = len(X_c) / n_samples
            self.means_[i] = X_c.mean(axis=0)
            centered = X_c - self.means_[i]
            cov = centered.T @ centered / len(X_c)
            self.covariances_[i] = cov

            stabilized_cov = cov + self.reg_covar * eye

            if self.reg_covar == 0.0 and np.linalg.matrix_rank(centered) < n_features:
                raise ValueError(
                    f"Maximum-likelihood covariance of class {c!r} is singular "
                    "or not positive definite; the unregularized multivariate "
                    "Gaussian density is undefined for this training fold."
                )

            try:
                self.cholesky_factors_[i] = np.linalg.cholesky(stabilized_cov)
            except np.linalg.LinAlgError as err:
                raise ValueError(
                    f"Maximum-likelihood covariance of class {c!r} is singular "
                    "or not positive definite; the multivariate Gaussian "
                    "density is undefined for this training fold."
                ) from err

        return self

    def _log_likelihood(self, X, mean, chol):
        """Compute log p(x | w_i) for every sample using a class Cholesky factor.

        Args:
            X (np.ndarray): Feature matrix of shape (n_samples, n_features).
            mean (np.ndarray): Class mean vector of shape (n_features,).
            chol (np.ndarray): Lower Cholesky factor L of shape
                (n_features, n_features) such that L @ L.T = Sigma.

        Returns:
            np.ndarray: Log-density values of shape (n_samples,).
        """
        n_features = X.shape[1]

        # Mahalanobis distance = ||z||^2, where L z = x - mu.
        z = solve_triangular(chol, (X - mean).T, lower=True)
        mahalanobis = np.sum(z ** 2, axis=0)

        # log|Sigma| = 2 * sum(log(diag(L))).
        log_det = 2.0 * np.sum(np.log(np.diag(chol)))

        return -0.5 * (n_features * np.log(2.0 * np.pi) + log_det + mahalanobis)

    def _joint_log_likelihood(self, X):
        """Compute log p(x | w_i) + log P(w_i) for every sample and class.

        Args:
            X (array-like of shape (n_queries, n_features)): Query samples.

        Returns:
            np.ndarray: Joint log-likelihood matrix of shape
                (n_queries, n_classes).
        """
        check_is_fitted(self)
        # reset=False checks that X has the same number of features as in fit.
        X = validate_data(self, X, dtype=float, reset=False)

        jll = np.empty((X.shape[0], len(self.classes_)))
        for i in range(len(self.classes_)):
            jll[:, i] = (
                self._log_likelihood(X, self.means_[i], self.cholesky_factors_[i])
                + np.log(self.priors_[i])
            )
        return jll

    def predict(self, X):
        """Predict the class with the highest posterior (Bayes' rule).

        The evidence is common to all classes, so maximizing log p(x | w_i) + log P(w_i) is enough.

        Args:
            X (array-like of shape (n_queries, n_features)): Test samples.

        Returns:
            numpy.ndarray: Predicted class labels.
        """
        jll = self._joint_log_likelihood(X)
        return self.classes_[np.argmax(jll, axis=1)]

    def predict_proba(self, X):
        """Return posteriors P(w_i | x), normalized with logsumexp to avoid underflow.

        Args:
            X (array-like of shape (n_queries, n_features)): Test samples.

        Returns:
            numpy.ndarray: Posterior probability estimates.
        """
        jll = self._joint_log_likelihood(X)
        return np.exp(jll - logsumexp(jll, axis=1, keepdims=True))

    @staticmethod
    def get_param_grid():
        """Return the empty hyperparameter grid for cross-validation.

        Returns:
            dict: Empty dictionary because the maximum-likelihood Gaussian model
                has no tunable hyperparameters.
        """
        return {}

from sklearn.base import BaseEstimator, ClassifierMixin
from sklearn.neighbors import KNeighborsClassifier
from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.utils.validation import check_is_fitted, validate_data


class BayesianKNNClassifier(ClassifierMixin, BaseEstimator):
    """Bayesian classifier based on k-nearest neighbors.

    This class wraps scikit-learn's KNeighborsClassifier. With uniform weights,
    k-NN density estimates and ML priors, the Bayes rule reduces to the
    majority vote among the k neighbors.

    Args:
        n_neighbors (int, optional): Number of neighbors to use. Defaults to 5.
        metric (str, optional): The distance metric to use. Defaults to 'euclidean'.
    """

    def __init__(self, n_neighbors=5, metric='euclidean'):
        self.n_neighbors = n_neighbors
        self.metric = metric

    def fit(self, X, y):
        """Fits the model based on the k-nearest neighbors.

        Args:
            X (array-like of shape (n_samples, n_features)): Training vectors.
            y (array-like of shape (n_samples,)): Target values.

        Returns:
            self: The fitted classifier.
        """
        X, y = validate_data(self, X, y)

        self.model_ = make_pipeline(
            StandardScaler(),
            KNeighborsClassifier(
                n_neighbors=self.n_neighbors,
                metric=self.metric
            )
        )
        self.model_.fit(X, y)
        self.classes_ = self.model_.classes_
        return self

    def predict(self, X):
        """Predicts the class labels for the provided data.

        Args:
            X (array-like of shape (n_queries, n_features)): Test samples.

        Returns:
            numpy.ndarray: Predicted class labels for each data sample.
        """
        check_is_fitted(self)
        X = validate_data(self, X, reset=False)
        return self.model_.predict(X)

    def predict_proba(self, X):
        """Returns the posterior probabilities for each class.

        The probabilities are calculated through the mathematical proportion k_i / k.

        Args:
            X (array-like of shape (n_queries, n_features)): Test samples.

        Returns:
            numpy.ndarray: Posterior probability estimates.
        """
        check_is_fitted(self)
        X = validate_data(self, X, reset=False)
        return self.model_.predict_proba(X)

    @staticmethod
    def get_param_grid():
        """Returns the hyperparameter search grid.

        Used to fix the number of neighbors (k) and the distance metric via
        cross-validation.

        Returns:
            dict: Dictionary containing the parameter grid for `n_neighbors`
            and `metric`.
        """
        return {
            'n_neighbors': [1, 3, 5, 7, 11, 15, 21, 31, 45, 63, 91],
            'metric': ['euclidean', 'cityblock', 'chebyshev']
        }

        self.solver = solver

from sklearn.base import BaseEstimator, ClassifierMixin
from sklearn.ensemble import VotingClassifier
from sklearn.utils.validation import check_is_fitted, validate_data


class MajorityVotingClassifier(ClassifierMixin, BaseEstimator):
    """Majority voting ensemble classifier.

    Combines the predictions of the Bayesian Gaussian, Bayesian k-NN,
    Parzen Window and Logistic Regression classifiers using hard voting.
    Ties are broken by the smallest class label (scikit-learn behavior).

    Args:
        estimators (list of tuples): List of (name, estimator) tuples for the
            base models.
    """

    def __init__(self, estimators):
        self.estimators = estimators

    def fit(self, X, y):
        """Fit the ensemble classifier using hard voting.

        Args:
            X (array-like of shape (n_samples, n_features)): Training vectors.
            y (array-like of shape (n_samples,)): Target values.

        Returns:
            self: The fitted classifier.
        """
        X, y = validate_data(self, X, y)
        self.model_ = VotingClassifier(estimators=self.estimators, voting='hard')
        self.model_.fit(X, y)
        self.classes_ = self.model_.classes_
        return self

    def predict(self, X):
        """Predict class labels for samples based on majority voting.

        Args:
            X (array-like of shape (n_queries, n_features)): Test samples.

        Returns:
            numpy.ndarray: Predicted class labels.
        """
        check_is_fitted(self)
        X = validate_data(self, X, reset=False)
        return self.model_.predict(X)

    @staticmethod
    def get_param_grid():
        """Return the parameter grid for hyperparameter tuning.

        MajorityVoting ensemble does not have hyperparameters to tune,
        as it uses fixed hard voting. The base estimators are tuned
        separately before being passed to the ensemble.

        Returns:
            dict: Empty parameter grid (no tuning needed).
        """
        return {}

import numpy as np
from sklearn.base import BaseEstimator, ClassifierMixin
from sklearn.neighbors import KernelDensity

class ParzenWindowClassifier(ClassifierMixin, BaseEstimator):
    """
    Bayesian classifier based on the Parzen window method using KDE.

    The classifier uses a multivariate product kernel composed of univariate
    Gaussian kernels.

    Args:
        bandwidth: Bandwidth used by each kernel density estimator.
    """
    def __init__(self, bandwidth=1.0):
        # Bandwidth h required by the project.
        self.bandwidth = bandwidth
        self.classes_ = None
        self.kdes_ = {}
        self.priors_ = {}

    def fit(self, X, y):
        """
        Fit a density estimator for each class and calculate prior probabilities.

        Args:
            X: Training samples.
            y: Target class labels.

        Returns:
            The fitted classifier.
        """
        self.classes_ = np.unique(y)
        n_total = len(y)
        
        for c in self.classes_:
            # Filter samples belonging to the current class.
            X_c = X[y == c]
            
            # Maximum likelihood estimate for P(w_i): P(w_i) = N_i / N_total.
            self.priors_[c] = len(X_c) / n_total
            
            # Fit the kernel density estimator for class c.
            kde = KernelDensity(bandwidth=self.bandwidth, kernel='gaussian')
            kde.fit(X_c)
            self.kdes_[c] = kde
            
        return self

    def predict(self, X):
        """
        Predict classes using Bayes' rule.

        The posterior probability is proportional to p(x | w_i) * P(w_i).

        Args:
            X: Samples to classify.

        Returns:
            Predicted class labels for each sample.
        """
        # Store the log probabilities for each class.
        log_probs = np.zeros((X.shape[0], len(self.classes_)))
        
        for i, c in enumerate(self.classes_):
            # score_samples returns the log density: log(p(x | w_i)).
            log_density = self.kdes_[c].score_samples(X)
            
            # Add the log prior to obtain the log posterior probability.
            # log(p(x | w_i) * P(w_i)) = log(p(x | w_i)) + log(P(w_i)).
            log_probs[:, i] = log_density + np.log(self.priors_[c])
            
        # Return the class with the highest posterior probability.
        return self.classes_[np.argmax(log_probs, axis=1)]

    @staticmethod
    def get_param_grid():
        """
        Return the hyperparameter grid for cross-validation searches.

        Returns:
            A dictionary containing the candidate hyperparameter values.
        """
        return {'bandwidth': [0.1, 0.5, 1.0, 2.0, 5.0, 10.0]}

## Decisões de Arquitetura: Double k-means

* **Algoritmo Seguido:** A classe `DoubleKMeans` implementa o double k-means (Vichi, 2001) conforme os slides da disciplina (`docs/Double k-means.pdf`). O algoritmo obtém ao mesmo tempo uma partição exclusiva dos objetos em $K$ grupos, uma partição exclusiva das variáveis em $H$ grupos e a matriz de protótipos $G = (g_{kh})$ dos $K \times H$ blocos. A função objetivo minimizada é $W(G, U, V) = \sum_{k=1}^{K}\sum_{h=1}^{H}\sum_{i=1}^{N}\sum_{j=1}^{P} u_{ik}\, v_{jh}\, (x_{ij} - g_{kh})^2$. A otimização alterna os três passos dos slides, sempre nessa ordem, até que nenhum objeto e nenhuma variável mude de grupo:
  1. com $U$ e $V$ fixos, $g_{kh}$ é a média do bloco $kh$: $g_{kh} = \frac{\sum_i \sum_j u_{ik} v_{jh} x_{ij}}{n_k\, n_h}$;
  2. com $G$ e $V$ fixos, cada objeto vai para o grupo $k$ que minimiza $\sum_h \sum_j v_{jh} (x_{ij} - g_{kh})^2$;
  3. com $G$ e $U$ fixos, cada variável vai para o grupo $h$ que minimiza $\sum_k \sum_i u_{ik} (x_{ij} - g_{kh})^2$.

  Como nos slides, $G$ não é recalculado entre os passos 2 e 3, a única exceção são os protótipos de grupos que ficaram vazios no passo 2 e foram reposicionados (ver *Tratamento de Grupos Vazios*). O parâmetro `max_iter` é apenas um limite de segurança, o atributo `converged_` indica se a parada ocorreu pelo critério dos slides.

  **Empates nos passos 2 e 3:** os slides definem $u_{ik} = 1$ (e $v_{jh} = 1$) para o grupo de custo mínimo, mas não dizem o que fazer quando dois grupos empatam. Por conseguinte, adotou-se a regra da melhora estrita: um objeto (ou variável) só muda de grupo se o custo do novo grupo, calculado com o $G$ fixo do passo, for menor que o do grupo atual por mais de $10^{-9}$, e em caso de empate, ele permanece onde está. A margem de $10^{-9}$ existe porque os custos são calculados em ponto flutuante: sem ela, uma diferença de arredondamento da ordem de $10^{-16}$ entre dois grupos empatados contaria como melhora e moveria o item sem ganho real em $W$. Por ser absoluta, a margem pressupõe a escala padronizada usada na Questão 1, em que o erro de arredondamento dos custos fica muito abaixo dela. Com variáveis de valores muito grandes, esse erro pode superá-la, e a proteção passa a ser o `max_iter`. É o mesmo princípio usado no TwoMP (Schepers & Hofmans, 2009) e no k-means de Hartigan & Wong (1979). Nesses dois, porém, cada transferência é avaliada individualmente pela variação exata da função objetivo, com os protótipos já atualizados, enquanto aqui a alocação é feita em lote contra $G$ fixo, como nos slides. Por isso, toda solução em que o Hartigan–Wong para também faz este algoritmo parar, mas não o contrário.

  Com essa regra, toda mudança de grupo feita pelos passos 2 e 3 reduz $W$ estritamente, e o reposicionamento de grupos vazios, que só ocorre depois de alguma mudança, não aumenta $W$. Assim, o valor de $W$ com os protótipos ótimos, que é o que o `objective_history_` registra, cai estritamente de uma partição para a seguinte. Como o número de partições é finito, em aritmética exata o algoritmo não revisita nenhuma partição e atinge a parada dos slides em um número finito de iterações. A margem de $10^{-9}$ impede que erros de arredondamento no cálculo dos custos criem transferências sem ganho real, e o `max_iter` limita o número de iterações caso isso ainda aconteça. Uma execução que o atinge termina com `converged_` falso.

* **Integração com o Ecossistema Scikit-Learn:** A classe herda de `ClusterMixin` e `BaseEstimator`, nessa ordem, pelo mesmo motivo explicado na seção do Bayesiano Gaussiano em `src/classification/README.md`. O `__init__` apenas guarda os parâmetros (`n_row_clusters` = $K$, `n_col_clusters` = $H$, `max_iter`, `random_state`), e a entrada é verificada com `validate_data`. Os parâmetros são validados no `fit`, que lança um `ValueError` se $K$ não for um inteiro entre 1 e $N$, se $H$ não for um inteiro entre 1 e $P$, ou se `max_iter` não for um inteiro maior ou igual a 1 (mesma restrição do `KMeans` do scikit-learn). As mensagens dos dois primeiros erros citam `n_samples = N` e `n_features = P`, no formato que os checks de estimador do scikit-learn (`parametrize_with_checks`) esperam; esses checks rodam em `tests/test_double_kmeans.py`. Depois do `fit`, os resultados ficam nos atributos:
  * `row_labels_` (também em `labels_`) e `column_labels_`: as partições de objetos ($U$) e de variáveis ($V$);
  * `prototypes_`: a matriz $G$, de tamanho $K \times H$;
  * `objective_`: o valor final de $W$;
  * `objective_history_`: o valor de $W$ na inicialização e ao fim de cada iteração, usado no plot da função objetivo versus as iterações. Cada ponto é $W(G, U, V)$ com os rótulos do fim da iteração e o $G$ recalculado para esses rótulos (a fórmula do passo 1), isto é, o menor $W$ possível para aquela partição. Esse $G$ serve só para o registro; a iteração seguinte o calcula de novo no seu passo 1. Assim, o último ponto é sempre igual a `objective_`, mesmo quando a execução para pelo `max_iter` (`converged_` falso), e o gráfico da Questão 1 não precisa de ponto à parte. Quando a execução converge, o último ponto repete o penúltimo, porque a última iteração não muda nenhum grupo;
  * `n_iter_`, `converged_` e `n_relocations_`.

* **Uma Execução por `fit`:** Cada chamada ao `fit` é uma execução a partir de uma inicialização aleatória, controlada por `random_state`. As 100 execuções por par $(K, H)$ exigidas na Questão 1, e a escolha da execução de menor $W$, ficam a cargo do script do experimento, que também precisa dos resultados de cada execução.

* **Sem Transformação na Classe:** A classe não transforma os dados: ela agrupa a matriz que recebe, como os estimadores do scikit-learn. A padronização das variáveis (z-score), decidida pelo grupo para a Questão 1, é feita no script do experimento (ver `src/experiments/README.md`).

* **Inicialização:** $U^{(0)}$ e $V^{(0)}$ são sorteados, atribuindo cada objeto e cada variável a um grupo aleatório. Para que nenhum grupo comece vazio, os $K$ (ou $H$) primeiros itens de uma permutação aleatória recebem um grupo diferente cada. $G^{(0)}$ é então obtido pelo passo 1.

* **Cálculo Matricial dos Passos:** Com $U$ ($N \times K$) e $V$ ($P \times H$) em formato *one-hot*, as somas dos blocos saem de $U^\top X V$. Os custos dos passos 2 e 3 saem da expansão $(x - g)^2 = x^2 - 2xg + g^2$, descartando o termo $x^2$, que não depende do grupo. Assim, cada passo é um pequeno número de produtos de matrizes, sem laços sobre objetos ou variáveis, o que importa para as 900 execuções da Questão 1.

* **Tratamento de Grupos Vazios:** Os slides não tratam desse caso. A inicialização garante grupos não vazios, mas os passos 2 e 3 podem esvaziar um grupo de objetos ($n_k = 0$) ou de variáveis ($n_h = 0$). Nesse caso, o passo 1 calcularia $0/0$, e o `NaN` resultante corromperia os passos seguintes sem gerar erro. A solução adotada é reposicionar: logo após os passos 2 e 3, cada grupo vazio recebe o objeto (ou a variável) de maior custo em relação ao próprio protótipo, escolhido entre os grupos com pelo menos dois membros. O protótipo do grupo que recebeu o item passa a ser a média desse item em cada grupo da outra partição. O mesmo método (`_relocate_empty_groups`) atende às duas partições: ele é escrito para os objetos, e as variáveis são tratadas como o problema transposto, passando $X^\top$ e $G^\top$.
  * **Justificativa:** o custo do item movido não pode aumentar, porque a média minimiza a soma de quadrados, ademais, os custos dos demais itens não mudam, e o grupo doador não fica vazio. Com isso, $W$ não aumenta entre iterações, e toda execução termina com exatamente $K$ grupos de objetos e $H$ grupos de variáveis. Nenhuma execução precisa ser descartada.
  * **Precedentes:** a mesma ideia de preencher o grupo vazio com o ponto de maior custo é usada pelo `KMeans` do scikit-learn (`_relocate_empty_clusters_dense`, em `sklearn/cluster/_k_means_common.pyx`). A implementação do double k-means publicada por Prunila & Vichi, o pacote R `drclust` (função `doublekm`), também preenche o grupo vazio logo após a alocação em vez de descartar a execução, com uma variante mais cara: divide em dois o grupo mais heterogêneo.
  * O atributo `n_relocations_` registra quantos reposicionamentos ocorreram em cada execução. Por segurança, o passo 1 levanta um `RuntimeError` se ainda assim encontrar um bloco vazio.

import numbers

import numpy as np
from sklearn.base import BaseEstimator, ClusterMixin
from sklearn.utils import check_random_state
from sklearn.utils.validation import validate_data


class DoubleKMeans(ClusterMixin, BaseEstimator):
    """Double k-means (Vichi, 2001): hard partition of objects and variables.

    Minimizes W(G, U, V) with the three steps of the course slides. Each `fit`
    is one run from a random start. Design decisions (ties, empty groups,
    stopping) are in src/clustering/README.md.

    Args:
        n_row_clusters (int, optional): Number of object groups K. Defaults to 2.
        n_col_clusters (int, optional): Number of variable groups H. Defaults to 2.
        max_iter (int, optional): Safety limit on iterations, must be >= 1.
            Defaults to 100.
        random_state (int, RandomState or None, optional): Seed of the random
            initial partitions. Defaults to None.
    """

    def __init__(self, n_row_clusters=2, n_col_clusters=2, max_iter=100,
                 random_state=None):
        self.n_row_clusters = n_row_clusters
        self.n_col_clusters = n_col_clusters
        self.max_iter = max_iter
        self.random_state = random_state

    def fit(self, X, y=None):
        """Run double k-means from a random initial partition.

        Args:
            X (array-like of shape (n_samples, n_features)): Data matrix.
            y: Ignored, present for API consistency.

        Returns:
            self: The fitted estimator.

        Raises:
            ValueError: If `n_row_clusters`, `n_col_clusters` or `max_iter`
                is not an integer in its valid range.
            RuntimeError: If a group is still empty when the prototypes are
                updated (a safety check; the relocation of empty groups
                should prevent it).
        """
        if not (isinstance(self.max_iter, numbers.Integral) and self.max_iter >= 1):
            raise ValueError(
                f"max_iter must be an integer >= 1, got {self.max_iter!r}."
            )

        X = validate_data(self, X, dtype=float)
        n_samples, n_features = X.shape
        K, H = self.n_row_clusters, self.n_col_clusters
        if not (isinstance(K, numbers.Integral) and 1 <= K <= n_samples):
            raise ValueError(f"n_row_clusters must be an integer in [1, "
                             f"n_samples = {n_samples}], got {K!r}.")
        if not (isinstance(H, numbers.Integral) and 1 <= H <= n_features):
            raise ValueError(f"n_col_clusters must be an integer in [1, "
                             f"n_features = {n_features}], got {H!r}.")

        rng = check_random_state(self.random_state)

        row_labels = self._random_partition(n_samples, K, rng)
        col_labels = self._random_partition(n_features, H, rng)
        G = self._update_prototypes(X, row_labels, col_labels, K, H)
        history = [self._objective(X, G, row_labels, col_labels)]

        converged = False
        n_relocations = 0
        for n_iter in range(1, self.max_iter + 1):
            # Step 1: prototypes.
            G = self._update_prototypes(X, row_labels, col_labels, K, H)

            # Step 2: objects (costs without the x_ij^2 term).
            col_onehot = np.eye(H)[col_labels]
            n_h = col_onehot.sum(axis=0)
            row_cost = -2.0 * (X @ col_onehot) @ G.T + (G ** 2) @ n_h
            new_row_labels = self._assign(row_cost, row_labels)
            new_row_labels, G, n_moved = self._relocate_empty_groups(
                X, G, new_row_labels, col_labels)
            n_relocations += n_moved

            # Step 3: variables, with the G of step 2.
            row_onehot = np.eye(K)[new_row_labels]
            n_k = row_onehot.sum(axis=0)
            col_cost = -2.0 * (row_onehot.T @ X).T @ G + n_k @ (G ** 2)
            new_col_labels = self._assign(col_cost, col_labels)
            new_col_labels, G_t, n_moved = self._relocate_empty_groups(
                X.T, G.T, new_col_labels, new_row_labels)
            G = G_t.T
            n_relocations += n_moved

            changed = (np.any(new_row_labels != row_labels)
                       or np.any(new_col_labels != col_labels))
            row_labels, col_labels = new_row_labels, new_col_labels

            # Update G for the current partition before saving it in the history.
            G_opt = self._update_prototypes(X, row_labels, col_labels, K, H)
            history.append(self._objective(X, G_opt, row_labels, col_labels))

            if not changed:
                converged = True
                break

        G = self._update_prototypes(X, row_labels, col_labels, K, H)

        self.row_labels_ = row_labels
        self.column_labels_ = col_labels
        self.labels_ = row_labels
        self.prototypes_ = G
        self.objective_ = self._objective(X, G, row_labels, col_labels)
        self.objective_history_ = np.array(history)
        self.n_iter_ = n_iter
        self.converged_ = converged
        self.n_relocations_ = n_relocations
        return self

    @staticmethod
    def _assign(cost, current_labels):
        """Move each item to its lowest-cost group only if its cost drops by
        more than 1e-9, so floating-point noise does not count as improvement."""
        best = np.argmin(cost, axis=1)
        items = np.arange(len(current_labels))
        improves = cost[items, current_labels] - cost[items, best] > 1e-9
        return np.where(improves, best, current_labels)

    @staticmethod
    def _relocate_empty_groups(X, G, labels, other_labels):
        """Refill groups emptied by steps 2 and 3.

        Written for objects; variables pass X.T and G.T.

        Returns:
            tuple: (labels, G, number of relocated items).
        """
        n_groups, n_other_groups = G.shape
        counts = np.bincount(labels, minlength=n_groups)
        empty_groups = np.flatnonzero(counts == 0)
        if empty_groups.size == 0:
            return labels, G, 0

        labels, G = labels.copy(), G.copy()
        other_sizes = np.bincount(other_labels, minlength=n_other_groups)
        for group in empty_groups:
            cost = np.sum((X - G[np.ix_(labels, other_labels)]) ** 2, axis=1)
            # Donors keep at least one member.
            cost[counts[labels] < 2] = -np.inf
            item = np.argmax(cost)

            counts[labels[item]] -= 1
            counts[group] += 1
            labels[item] = group
            G[group] = np.bincount(other_labels, weights=X[item],
                                   minlength=n_other_groups) / other_sizes

        return labels, G, empty_groups.size

    @staticmethod
    def _random_partition(n_items, n_groups, rng):
        """Assign items to groups at random, with no empty group."""
        labels = rng.randint(n_groups, size=n_items)
        labels[rng.permutation(n_items)[:n_groups]] = np.arange(n_groups)
        return labels

    @staticmethod
    def _update_prototypes(X, row_labels, col_labels, K, H):
        """Step 1: g_kh = sum of block kh / (n_k * n_h)."""
        row_onehot = np.eye(K)[row_labels]
        col_onehot = np.eye(H)[col_labels]
        block_sums = row_onehot.T @ X @ col_onehot
        block_sizes = np.outer(row_onehot.sum(axis=0), col_onehot.sum(axis=0))

        # Safety check: an empty block would put NaN in G.
        if np.any(block_sizes == 0):
            raise RuntimeError("Empty group in the prototype update.")

        return block_sums / block_sizes

    @staticmethod
    def _objective(X, G, row_labels, col_labels):
        """W(G, U, V): squared distance of each x_ij to its block prototype."""
        return float(np.sum((X - G[np.ix_(row_labels, col_labels)]) ** 2))

## Protocolo

- Grade fixa: `K` em `{2, 3, 4}`, `H` de `1` ate `K` (9 pares).
- Sementes: `random_state=run`, de `0` a `n_runs - 1`, em cada par.
- Vencedor de cada par: menor funcao objetivo; em empate exato, menor `run`.
- Execucoes sem convergencia sao mantidas, com aviso explicito. Consulte
  `convergence_rate` e `best_converged` antes de interpretar os resultados.
- O checkpoint e salvo atomicamente ao terminar cada par. Uma interrupcao
  durante um par exige repetir somente esse par; os anteriores sao reutilizados.
- Sem `--resume`, um checkpoint existente causa erro, evitando sobrescrita
  acidental. Para mudar parametros, use outro `--output-dir`.

- `q1_executions_summary.csv`: uma linha por par concluido, com `K`, `H`,
  `n_runs`, `best_objective`, `best_run_id`, `n_iterations`, `convergence_rate`
  (fracao de 0 a 1) e `best_converged`. O experimento completo tem 9 linhas.

DATASET_NAMES = ("original_2_classes", "kstar_classes")

)
DEFAULT_TRAIN_SIZES = tuple(np.arange(0.05, 0.96, 0.05).round(2))


def _make_classifiers():
    """Create fresh base estimators before tuning each learning-curve split.

    Returns:
        list[tuple[str, BaseEstimator]]: List of the four base
            ``(name, estimator)`` pairs.
    """
    base_estimators = [
        ("GaussianBayes", GaussianBayesClassifier()),
        ("BayesianKNN", BayesianKNNClassifier()),
        ("ParzenWindow", ParzenWindowClassifier()),
        ("LogisticRegression", CustomLogisticRegression()),
    ]
    return base_estimators


def _tune_classifiers(
    X_train: np.ndarray,
    y_train: np.ndarray,
    random_state: int,
    n_jobs: int = 1,
) -> tuple[list[tuple[str, BaseEstimator]], dict[str, dict[str, object]]]:
    """Tune base classifiers with stratified CV using training data only.

    Args:
        X_train (np.ndarray): Training feature matrix for the current split.
        y_train (np.ndarray): Training target labels for the current split.
        random_state (int): Seed for the inner ``StratifiedKFold`` splitter.
        n_jobs (int, optional): Number of parallel jobs for ``GridSearchCV``.
            Defaults to 1.

    Returns:
        tuple[list[tuple[str, BaseEstimator]], dict[str, dict[str, object]]]:
            List of fitted ``(name, best_estimator)`` tuples and a dictionary of
            selected hyperparameters per classifier.

    Raises:
        ValueError: If the rarest class has fewer than 2 training examples or
            if no valid hyperparameter candidates remain for a classifier.
    """
    class_counts = np.unique(y_train, return_counts=True)[1]
    minimum_class_count = int(class_counts.min())
    if minimum_class_count < 2:
        raise ValueError(
            "At least two training examples per class are needed for stratified "
            "cross-validation; increase the learning-curve training proportion."
        )

    n_splits = min(5, minimum_class_count)
    inner_cv = StratifiedKFold(
        n_splits=n_splits,
        shuffle=True,
        random_state=random_state,
    )
    inner_splits = list(inner_cv.split(X_train, y_train))
    minimum_inner_train_size = min(len(train_indices) for train_indices, _ in inner_splits)
    tuned_estimators = []
    selected_params = {}

    for classifier_name, classifier in _make_classifiers():
        parameter_grid = classifier.get_param_grid()
        if classifier_name == "BayesianKNN":
            parameter_grid["n_neighbors"] = [
                neighbors
                for neighbors in parameter_grid["n_neighbors"]
                if neighbors <= minimum_inner_train_size
            ]
        if any(not values for values in parameter_grid.values()):
            raise ValueError(
                f"No valid hyperparameter candidates remain for {classifier_name} "
                f"with the current training subset."
            )

        search = GridSearchCV(
            estimator=classifier,
            param_grid=parameter_grid,
            scoring="f1_macro",
            cv=inner_splits,
            refit=True,
            error_score="raise",
            n_jobs=n_jobs,
        )
        search.fit(X_train, y_train)
        tuned_estimators.append((classifier_name, search.best_estimator_))
        selected_params[classifier_name] = search.best_params_

    return tuned_estimators, selected_params


def _evaluate_learning_curve_split(
    dataset_name: str,
    train_size: float,
    repeat_index: int,
    train_indices: np.ndarray,
    test_indices: np.ndarray,
    X: np.ndarray,
    y: np.ndarray,
    random_state: int,
) -> list[dict[str, object]]:
    """Evaluate all 5 classifiers on a single (train_size, repeat) split.

    Args:
        dataset_name (str): Name of the dataset being evaluated.
        train_size (float): Proportion of data used for training in [0.05, 0.95].
        repeat_index (int): 1-based index of the stratified split repetition.
        train_indices (np.ndarray): Row indices of the training set.
        test_indices (np.ndarray): Row indices of the test set.
        X (np.ndarray): Full feature matrix.
        y (np.ndarray): Full target vector.
        random_state (int): Base random seed for inner CV splitting.

    Returns:
        list[dict[str, object]]: List of 5 evaluation records (one per
            classifier) with train and test macro-F1 scores.
    """
    X_train, y_train = X[train_indices], y[train_indices]
    X_test, y_test = X[test_indices], y[test_indices]

    tuned_estimators, selected_params = _tune_classifiers(
        X_train,
        y_train,
        random_state=random_state + repeat_index,
        n_jobs=1,
    )
    voting_estimators = [
        (name, clone(estimator)) for name, estimator in tuned_estimators
    ]
    classifiers = tuned_estimators + [
        (
            "MajorityVoting",
            MajorityVotingClassifier(estimators=voting_estimators),
        )
    ]
    selected_params["MajorityVoting"] = {
        name: params for name, params in selected_params.items()
    }
    inner_cv_splits = min(
        5, int(np.unique(y_train, return_counts=True)[1].min())
    )

    split_rows = []
    for classifier_name, classifier in classifiers:
        if classifier_name == "MajorityVoting":
            classifier.fit(X_train, y_train)
        train_prediction = classifier.predict(X_train)
        test_prediction = classifier.predict(X_test)
        split_rows.append(
            {
                "dataset": dataset_name,
                "classifier": classifier_name,
                "train_size": train_size,
                "repeat": repeat_index,
                "n_train": len(train_indices),
                "n_test": len(test_indices),
                "inner_cv_splits": inner_cv_splits,
                "best_params": json.dumps(
                    selected_params[classifier_name],
                    sort_keys=True,
                ),
                "train_f1": f1_score(
                    y_train,
                    train_prediction,
                    average="macro",
                    zero_division=0,
                ),
                "test_f1": f1_score(
                    y_test,
                    test_prediction,
                    average="macro",
                    zero_division=0,
                ),
            }
        )
    return split_rows


def _summarize_learning_curve_results(raw_results: pd.DataFrame) -> pd.DataFrame:
    """Aggregate split-level scores into means and standard deviations.

    Args:
        raw_results (pd.DataFrame): Split-level DataFrame produced by
            ``run_learning_curve_experiment``.

    Returns:
        pd.DataFrame: Summary DataFrame grouped by ``dataset``, ``classifier``,
            and ``train_size``.
    """
    return (
        raw_results.groupby(
            ["dataset", "classifier", "train_size"], as_index=False, sort=True
        )
        .agg(
            n_train=("n_train", "mean"),
            n_test=("n_test", "mean"),
            n_repeats=("repeat", "nunique"),
            train_f1_mean=("train_f1", "mean"),
            train_f1_std=("train_f1", "std"),
            test_f1_mean=("test_f1", "mean"),
            test_f1_std=("test_f1", "std"),
        )
        .fillna({"train_f1_std": 0.0, "test_f1_std": 0.0})
    )


def _plot_dataset_learning_curves(
    summary: pd.DataFrame,
    dataset_name: str,
    output_path: Path,
) -> None:
    """Plot train/test macro-F1 with ±1 SD for every classifier.

    Args:
        summary (pd.DataFrame): Aggregated learning-curve statistics.
        dataset_name (str): Name of the dataset to plot.
        output_path (Path): Destination PNG file path.
    """
    figure, axes = plt.subplots(1, len(CLASSIFIER_NAMES), figsize=(22, 5), sharey=True)

    for axis, classifier_name in zip(axes, CLASSIFIER_NAMES):
        classifier = summary[
            (summary["dataset"] == dataset_name)
            & (summary["classifier"] == classifier_name)
        ].sort_values("train_size")
        x_values = classifier["train_size"].to_numpy() * 100

        for score_name, color, label in (
            ("train", "#2878B5", "Treino"),
            ("test", "#D9534F", "Teste"),
        ):
            means = classifier[f"{score_name}_f1_mean"].to_numpy()
            stds = classifier[f"{score_name}_f1_std"].to_numpy()
            axis.plot(x_values, means, marker="o", markersize=3.5, color=color, label=label)
            axis.fill_between(
                x_values,
                np.clip(means - stds, 0, 1),
                np.clip(means + stds, 0, 1),
                color=color,
                alpha=0.16,
                linewidth=0,
            )

        axis.set_title(classifier_name, fontsize=10)
        axis.set_xlabel("Proporção de treino (%)")
        axis.set_xticks(np.arange(5, 100, 10))
        axis.set_xlim(2, 98)
        axis.set_ylim(0, 1.02)
        axis.grid(alpha=0.25, linestyle="--")
        axis.legend(frameon=False, fontsize=8)

    axes[0].set_ylabel("F1 macro")
    figure.suptitle(
        f"Curvas de aprendizagem — {dataset_name.replace('_', ' ')}\n"
        "Média ± 1 desvio-padrão em 10 divisões estratificadas",
        fontsize=13,
    )
    figure.tight_layout(rect=(0, 0, 1, 0.89), w_pad=1.0)
    figure.savefig(output_path, dpi=220, bbox_inches="tight")
    plt.close(figure)


def run_learning_curve_experiment(
    results_dir: str | Path = "results",
    plots_dir: str | Path = "plots",
    n_repeats: int = 10,
    random_state: int = 42,
    n_jobs: int = 1,
) -> tuple[pd.DataFrame, pd.DataFrame]:
    """Run and save learning curves for original and Q1-cluster datasets.

    Each base classifier is tuned by stratified CV using only the current
    training subset. The majority-voting model uses those selected parameters.

    Args:
        results_dir (str or Path, optional): Directory where detailed and
            summary CSVs are saved. Defaults to "results".
        plots_dir (str or Path, optional): Directory where learning-curve PNG
            plots are saved. Defaults to "plots".
        n_repeats (int, optional): Number of stratified splits per training
            proportion. Must be at least 2. Defaults to 10.
        random_state (int, optional): Base random seed for reproducible splits.
            Defaults to 42.
        n_jobs (int, optional): Number of parallel jobs across splits. Defaults
            to 1.

    Returns:
        tuple[pd.DataFrame, pd.DataFrame]: A tuple ``(raw_results, summary)``
            containing split-level scores and aggregated mean/SD statistics.

    Raises:
        ValueError: If ``n_repeats < 2``.
        RuntimeError: If the number of collected rows does not match the
            expected total.
    """
    if n_repeats < 2:
        raise ValueError("n_repeats must be at least 2 to estimate error bars.")

    results_dir = Path(results_dir)
    plots_dir = Path(plots_dir)
    results_dir.mkdir(parents=True, exist_ok=True)
    plots_dir.mkdir(parents=True, exist_ok=True)

    original_X, original_y = load_spambase_original()
    kstar_X, kstar_y = load_spambase_kstar_real()
    datasets = {
        "original_2_classes": (original_X, original_y),
        "kstar_classes": (kstar_X, kstar_y),
    }

    raw_rows = []
    for dataset_name, (X, y) in datasets.items():
        LOGGER.info(
            "Learning curves for %s: %d samples, %d features, class counts=%s",
            dataset_name,
            X.shape[0],
            X.shape[1],
            np.bincount(y).tolist(),
        )
        splits_by_size = get_learning_curve_splits(
            X,
            y,
            n_repeats=n_repeats,
            random_state=random_state,
        )

        for train_size in DEFAULT_TRAIN_SIZES:
            splits = splits_by_size[train_size]
            LOGGER.info(
                "  %s: train size %.0f%% (%d/%d splits)",
                dataset_name,
                train_size * 100,
                len(splits),
                n_repeats,
            )
            if n_jobs == 1:
                for repeat_index, (train_indices, test_indices) in enumerate(
                    splits, start=1
                ):
                    raw_rows.extend(
                        _evaluate_learning_curve_split(
                            dataset_name,
                            train_size,
                            repeat_index,
                            train_indices,
                            test_indices,
                            X,
                            y,
                            random_state,
                        )
                    )
            else:
                split_batches = Parallel(n_jobs=n_jobs)(
                    delayed(_evaluate_learning_curve_split)(
                        dataset_name,
                        train_size,
                        repeat_index,
                        train_indices,
                        test_indices,
                        X,
                        y,
                        random_state,
                    )
                    for repeat_index, (train_indices, test_indices) in enumerate(
                        splits, start=1
                    )
                )
                for split_rows in split_batches:
                    raw_rows.extend(split_rows)

    raw_results = pd.DataFrame(raw_rows)
    expected_rows = (
        len(DATASET_NAMES)
        * len(CLASSIFIER_NAMES)
        * len(DEFAULT_TRAIN_SIZES)
        * n_repeats
    )
    if len(raw_results) != expected_rows:
        raise RuntimeError(
            f"Expected {expected_rows} learning-curve rows, got {len(raw_results)}."
        )

    summary = _summarize_learning_curve_results(raw_results)
    raw_path = results_dir / "learning_curves_data.csv"
    summary_path = results_dir / "learning_curves_summary.csv"
    raw_results.to_csv(raw_path, index=False)
    summary.to_csv(summary_path, index=False)
    LOGGER.info("Split-level scores saved to %s", raw_path)
    LOGGER.info("Mean and standard-deviation summary saved to %s", summary_path)

    for dataset_name, plot_name in (
        ("original_2_classes", "learning_curves_original.png"),
        ("kstar_classes", "learning_curves_kstar.png"),
    ):
        plot_path = plots_dir / plot_name
        _plot_dataset_learning_curves(summary, dataset_name, plot_path)
        LOGGER.info("Learning-curve plot saved to %s", plot_path)

    return raw_results, summary


def main() -> None:
    """Parse CLI arguments and run the Question 2 learning-curve experiment."""
    parser = argparse.ArgumentParser(
        description="Generate learning curves for the two Question 2 datasets."
    )
    parser.add_argument("--results-dir", default="results")
    parser.add_argument("--plots-dir", default="plots")
    parser.add_argument("--n-repeats", type=int, default=10)
    parser.add_argument("--random-state", type=int, default=42)
    parser.add_argument(
        "--n-jobs",
        type=int,
        default=-1,
        help="Number of parallel jobs for GridSearchCV (default: -1, all CPU cores)",
    )
    args = parser.parse_args()

    logging.basicConfig(
        level=logging.INFO,
        format="%(asctime)s [%(levelname)s] %(message)s",
    )
    raw_results, _ = run_learning_curve_experiment(
        results_dir=args.results_dir,
        plots_dir=args.plots_dir,
        n_repeats=args.n_repeats,
        random_state=args.random_state,
        n_jobs=args.n_jobs,
    )
    LOGGER.info(
        "Learning-curve experiment completed: %d classifier evaluations.",
        len(raw_results),
    )


if __name__ == "__main__":
    main()

import argparse
from contextlib import contextmanager
import hashlib
import logging
import numbers
import os
from pathlib import Path
import pickle
import tempfile

import numpy as np
import pandas as pd
import sklearn
from sklearn.preprocessing import StandardScaler

from src.clustering.double_kmeans import DoubleKMeans
from src.utils.dataloader import load_spambase_original


Q1_GRID = tuple((K, H) for K in (2, 3, 4) for H in range(1, K + 1))
LOGGER = logging.getLogger(__name__)


@contextmanager
def _atomic_output(path):
    """Write binary output to a temporary file and atomically replace destination.

    Args:
        path (Path): Destination file path.

    Yields:
        IO[bytes]: Open binary file stream for writing.
    """
    with tempfile.NamedTemporaryFile(
        mode="wb", dir=path.parent, prefix=f".{path.name}.", delete=False
    ) as stream:
        temporary_path = Path(stream.name)
        try:
            yield stream
            stream.flush()
            os.fsync(stream.fileno())
            stream.close()
            os.replace(temporary_path, path)
        finally:
            temporary_path.unlink(missing_ok=True)


def _save_outputs(output_dir, history):
    """Persist per-pair best NPZ artifacts and the summary CSV.

    Args:
        output_dir (Path): Directory where summary and NPZ files are written.
        history (dict): Experiment history dictionary containing ``runs``.

    Returns:
        pd.DataFrame: Summary table with one row per completed ``(K, H)`` pair.
    """
    rows = []
    for K, H in Q1_GRID:
        runs = [run for run in history["runs"] if (run["K"], run["H"]) == (K, H)]
        if not runs:
            continue
        # min keeps the first run in case of an exact objective tie.
        best = min(runs, key=lambda run: run["objective"])
        rows.append({
            "K": K,
            "H": H,
            "n_runs": len(runs),
            "best_objective": best["objective"],
            "best_run_id": best["run"],
            "n_iterations": best["n_iter"],
            "convergence_rate": sum(run["converged"] for run in runs) / len(runs),
            "best_converged": best["converged"],
        })
        with _atomic_output(output_dir / f"q1_best_dkm_K{K}_H{H}.npz") as stream:
            np.savez_compressed(stream, **best)

    summary = pd.DataFrame(rows, columns=[
        "K", "H", "n_runs", "best_objective", "best_run_id", "n_iterations",
        "convergence_rate", "best_converged",
    ])
    with _atomic_output(output_dir / "q1_executions_summary.csv") as stream:
        stream.write(summary.to_csv(index=False).encode("utf-8"))
    return summary


def run_question_1(
    filepath="dataset/spambase.data",
    output_dir="results",
    n_runs=100,
    max_iter=100,
    resume=False,
):
    """Run 100 seeded starts per (K, H), checkpointing each completed pair.

    Features are standardized with a single z-score transform on the full Q1
    dataset. Original class labels are never passed to the clustering model.
    Resume accepts only a checkpoint with the same data and configuration.
    Existing checkpoints are not overwritten unless ``resume=True``.

    Args:
        filepath (str or Path, optional): Path to the Spambase dataset file.
            Defaults to "dataset/spambase.data".
        output_dir (str or Path, optional): Output directory for checkpoints and
            summaries. Defaults to "results".
        n_runs (int, optional): Number of seeded runs per (K, H) pair. Defaults
            to 100.
        max_iter (int, optional): Maximum iterations per Double k-means run.
            Defaults to 100.
        resume (bool, optional): Whether to resume from an existing checkpoint.
            Defaults to False.

    Returns:
        pd.DataFrame: Summary DataFrame with the minimum-objective run and
            convergence rate per (K, H) pair.

    Raises:
        ValueError: If ``n_runs`` or ``max_iter`` is invalid, if features are
            non-finite or too small, or if checkpoint metadata mismatches.
        FileExistsError: If a checkpoint already exists and ``resume=False``.
        FileNotFoundError: If ``resume=True`` and no checkpoint exists.
        RuntimeError: If a Double k-means run produces a non-finite objective.
    """
    for name, value in (("n_runs", n_runs), ("max_iter", max_iter)):
        if isinstance(value, bool) or not isinstance(value, numbers.Integral) or value < 1:
            raise ValueError(f"{name} must be an integer >= 1, got {value!r}.")

    filepath, output_dir = Path(filepath), Path(output_dir)
    history_path = output_dir / "q1_full_history.pkl"
    if history_path.exists() and not resume:
        raise FileExistsError(
            f"{history_path} already exists. Use --resume or a new output directory."
        )
    if resume and not history_path.exists():
        raise FileNotFoundError(f"No Q1 checkpoint found at {history_path}.")

    X, _ = load_spambase_original(filepath)
    if X.shape[0] < 4 or X.shape[1] < 4 or not np.isfinite(X).all():
        raise ValueError("Q1 requires finite features and at least 4 rows and 4 columns.")
    scaler = StandardScaler()
    X = scaler.fit_transform(X)
    metadata = {
        "format_version": 1,
        "dataset_sha256": hashlib.sha256(filepath.read_bytes()).hexdigest(),
        "shape": X.shape,
        "n_runs": int(n_runs),
        "max_iter": int(max_iter),
        "grid": Q1_GRID,
        "preprocessing": "StandardScaler",
        "numpy_version": np.__version__,
        "pandas_version": pd.__version__,
        "sklearn_version": sklearn.__version__,
    }
    if resume:
        # Only load checkpoints produced locally by this experiment.
        with history_path.open("rb") as stream:
            history = pickle.load(stream)
        if history["metadata"] != metadata:
            raise ValueError("Checkpoint data or configuration differs from this run.")
        expected = [(K, H, run) for K, H in Q1_GRID for run in range(n_runs)]
        actual = [(run["K"], run["H"], run["run"]) for run in history["runs"]]
        if len(actual) % n_runs or actual != expected[:len(actual)]:
            raise ValueError("Checkpoint must contain an ordered prefix of complete pairs.")
        _save_outputs(output_dir, history)
    else:
        history = {
            "metadata": metadata,
            "scaler_mean": scaler.mean_,
            "scaler_scale": scaler.scale_,
            "runs": [],
        }
        output_dir.mkdir(parents=True, exist_ok=True)
        with _atomic_output(history_path) as stream:
            pickle.dump(history, stream, protocol=pickle.HIGHEST_PROTOCOL)

    completed_pairs = len(history["runs"]) // n_runs
    for K, H in Q1_GRID[completed_pairs:]:
        for run in range(n_runs):
            model = DoubleKMeans(
                n_row_clusters=K,
                n_col_clusters=H,
                max_iter=max_iter,
                random_state=run,
            ).fit(X)
            if not np.isfinite(model.objective_):
                raise RuntimeError(f"Non-finite objective at K={K}, H={H}, run={run}.")
            history["runs"].append({
                "run": run,
                "K": K,
                "H": H,
                "objective": model.objective_,
                "n_iter": model.n_iter_,
                "converged": model.converged_,
                "n_relocations": model.n_relocations_,
                "labels": model.labels_.copy(),
                "column_labels": model.column_labels_.copy(),
                "prototypes": model.prototypes_.copy(),
                "objective_history": model.objective_history_.copy(),
            })
            LOGGER.info(
                "K=%d H=%d run=%d/%d objective=%.10g iterations=%d",
                K, H, run + 1, n_runs, model.objective_, model.n_iter_,
            )
            if not model.converged_:
                LOGGER.warning(
                    "K=%d H=%d seed=%d reached max_iter=%d without convergence.",
                    K, H, run, max_iter,
                )

        with _atomic_output(history_path) as stream:
            pickle.dump(history, stream, protocol=pickle.HIGHEST_PROTOCOL)
        _save_outputs(output_dir, history)
        LOGGER.info("Checkpoint saved for K=%d H=%d.", K, H)

    return pd.read_csv(output_dir / "q1_executions_summary.csv")


def main(argv=None):
    """Parse CLI arguments and execute the Question 1 experiment.

    Args:
        argv (list[str] or None, optional): Command-line argument list. Defaults

    """
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--data", default="dataset/spambase.data")
    parser.add_argument("--output-dir", default="results")
    parser.add_argument("--n-runs", type=int, default=100)
    parser.add_argument("--max-iter", type=int, default=100)
    parser.add_argument("--resume", action="store_true")
    args = parser.parse_args(argv)
    logging.basicConfig(level=logging.INFO, format="%(levelname)s: %(message)s")

"""Question 2: 30×10 nested cross-validation with 5 classifiers on 2 datasets.

This module orchestrates the nested cross-validation pipeline:
- Outer loop: 30 repetitions × 10 folds = 300 test sets
- Inner loop: 5-fold stratified CV for hyperparameter tuning
- Classifiers: Bayesian, Bayesian k-NN, Parzen, Logistic Regression, Majority Voting
- Datasets: Original (spam/non-spam) and K*-clustering (from Q1)
- Metrics: Error, Precision (macro), Recall (macro), F1-measure (macro)

Output: CSV with 3,000 rows (300 outer folds × 2 datasets × 5 classifiers).
Majority voting uses the individually tuned base estimators from the inner CV.
"""

import logging
import numpy as np
import pandas as pd
from pathlib import Path
from joblib import Parallel, delayed
from sklearn.base import clone
from sklearn.model_selection import GridSearchCV, StratifiedKFold
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score

from src.utils.dataloader import (
    load_spambase_original,
    load_spambase_kstar_real,
    get_nested_cv_splitters,
)
from src.classification.bayes_gaussian import GaussianBayesClassifier
from src.classification.bayesian_knn import BayesianKNNClassifier
from src.classification.parzen_window import ParzenWindowClassifier
from src.classification.custom_logistic_regression import CustomLogisticRegression
from src.classification.majority_voting import MajorityVotingClassifier


logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(message)s",
)
logger = logging.getLogger(__name__)


def _fit_tuned_majority_voting(X_train, y_train, inner_cv, n_jobs):
    """Tune each voting member on inner CV, then fit the ensemble on all training data.

    Args:
        X_train (np.ndarray): Training feature matrix of shape
            (n_train, n_features).
        y_train (np.ndarray): Training target labels of shape (n_train,).
        inner_cv (StratifiedKFold): Inner cross-validation splitter.
        n_jobs (int): Number of parallel workers for ``GridSearchCV``.

    Returns:
        tuple[MajorityVotingClassifier, dict[str, dict]]: The fitted majority
            voting classifier and a dictionary mapping each base classifier
            name to its selected hyperparameter dictionary.

    Raises:
        ValueError: If no valid hyperparameter candidates remain for a base
            classifier.
    """
    base_classifiers = [
        ("GaussianBayes", GaussianBayesClassifier()),
        ("BayesianKNN", BayesianKNNClassifier()),
        ("ParzenWindow", ParzenWindowClassifier()),
        ("LogisticRegression", CustomLogisticRegression()),
    ]
    cv_splits = list(inner_cv.split(X_train, y_train))
    minimum_inner_train_size = min(len(train_indices) for train_indices, _ in cv_splits)
    selected_estimators = []
    selected_params = {}

    for classifier_name, classifier in base_classifiers:
        param_grid = classifier.get_param_grid()
        if classifier_name == "BayesianKNN":
            param_grid["n_neighbors"] = [
                neighbors
                for neighbors in param_grid["n_neighbors"]
                if neighbors <= minimum_inner_train_size
            ]
        if any(not values for values in param_grid.values()):
            raise ValueError(
                f"No valid hyperparameter candidates remain for {classifier_name}."
            )

        grid_search = GridSearchCV(
            estimator=classifier,
            param_grid=param_grid,
            cv=cv_splits,
            scoring="f1_macro",
            n_jobs=n_jobs,
            error_score="raise",
        )
        grid_search.fit(X_train, y_train)
        selected_estimators.append(
            (classifier_name, clone(grid_search.best_estimator_))
        )
        selected_params[classifier_name] = grid_search.best_params_

    majority_voting = MajorityVotingClassifier(estimators=selected_estimators)
    majority_voting.fit(X_train, y_train)
    return majority_voting, selected_params


def _evaluate_outer_fold(
    dataset_name,
    fold_idx,
    train_idx,
    test_idx,
    X,
    y,
    inner_cv,
    classifiers_config,
    base_classifiers_config,
):
    """Evaluate all 5 classifiers on a single outer fold.

    Tunes the four base classifiers once via inner cross-validation and reuses
    their best estimators to assemble the ``MajorityVotingClassifier``.

    Args:
        dataset_name (str): Identifier of the evaluated dataset.
        fold_idx (int): Sequential outer fold index in [0, 299].
        train_idx (np.ndarray): Training row indices for this outer fold.
        test_idx (np.ndarray): Test row indices for this outer fold.
        X (np.ndarray): Full dataset feature matrix.
        y (np.ndarray): Full dataset target vector.
        inner_cv (StratifiedKFold): Inner 5-fold stratified splitter.
        classifiers_config (list[tuple[str, BaseEstimator]]): Full list of 5
            ``(name, estimator)`` pairs.
        base_classifiers_config (list[tuple[str, BaseEstimator]]): List of the
            first 4 base ``(name, estimator)`` pairs.

    Returns:
        list[dict]: List of 5 result dictionaries (one per classifier) for the
            given outer fold.
    """
    X_train_outer = X[train_idx]
    y_train_outer = y[train_idx]
    X_test = X[test_idx]
    y_test = y[test_idx]

    repetition = fold_idx // 10
    fold_in_rep = fold_idx % 10

    cv_splits = list(inner_cv.split(X_train_outer, y_train_outer))
    minimum_inner_train_size = min(
        len(train_indices) for train_indices, _ in cv_splits
    )
    tuned_estimators = []
    tuned_params = {}
    fold_results = []

    for clf_name, clf in classifiers_config:
        if clf_name == "MajorityVoting":
            if len(tuned_estimators) == len(base_classifiers_config):
                best_clf = MajorityVotingClassifier(
                    estimators=[
                        (name, clone(est)) for name, est in tuned_estimators
                    ]
                )
                best_clf.fit(X_train_outer, y_train_outer)
                best_params = dict(tuned_params)
            else:
                best_clf, best_params = _fit_tuned_majority_voting(
                    X_train_outer,
                    y_train_outer,
                    inner_cv,
                    n_jobs=1,
                )
            best_f1_inner = np.nan
        else:
            param_grid = clf.get_param_grid()
            if clf_name == "BayesianKNN":
                param_grid["n_neighbors"] = [
                    neighbors
                    for neighbors in param_grid["n_neighbors"]
                    if neighbors <= minimum_inner_train_size
                ]
            grid_search = GridSearchCV(
                estimator=clf,
                param_grid=param_grid,
                cv=cv_splits,
                scoring="f1_macro",
                n_jobs=1,
                verbose=0,
                error_score="raise",
            )
            grid_search.fit(X_train_outer, y_train_outer)
            best_params = grid_search.best_params_
            best_clf = grid_search.best_estimator_
            best_f1_inner = grid_search.best_score_
            tuned_estimators.append((clf_name, clone(best_clf)))
            tuned_params[clf_name] = best_params

        y_pred = best_clf.predict(X_test)

        error = 1.0 - accuracy_score(y_test, y_pred)
        precision = precision_score(
            y_test, y_pred, average="macro", zero_division=0
        )
        recall = recall_score(y_test, y_pred, average="macro", zero_division=0)
        f1 = f1_score(y_test, y_pred, average="macro", zero_division=0)

        fold_results.append(
            {
                "fold_id": fold_idx,
                "repetition": repetition + 1,
                "fold_in_rep": fold_in_rep + 1,
                "dataset": dataset_name,
                "classifier": clf_name,
                "error": error,
                "precision": precision,
                "recall": recall,
                "f1": f1,
                "best_params": str(best_params),
                "n_train": len(train_idx),
                "n_test": len(test_idx),
                "best_f1_inner": best_f1_inner,
            }
        )

    return fold_results


def run_nested_cv_experiment(
    output_dir="results",
    n_jobs=1,
    verbose=0,
):
    """Run 30×10 nested cross-validation on both datasets with 5 classifiers.

    Args:
        output_dir (str or Path, optional): Directory where results will be
            saved. Defaults to "results".
        n_jobs (int, optional): Number of parallel jobs across outer folds.
            Defaults to 1.
        verbose (int, optional): Verbosity level (0=silent, 1=progress,
            2=detailed). Defaults to 0.

    Returns:
        pd.DataFrame: DataFrame with 3,000 rows (300 folds × 2 datasets × 5
            classifiers) and columns ``fold_id``, ``repetition``,
            ``fold_in_rep``, ``dataset``, ``classifier``, ``error``,
            ``precision``, ``recall``, ``f1``, ``best_params``, ``n_train``,
            ``n_test``, and ``best_f1_inner``.
    """
    output_dir = Path(output_dir)
    output_dir.mkdir(parents=True, exist_ok=True)

    logger.info("=" * 70)
    logger.info("QUESTION 2: 30×10 Nested Cross-Validation Experiment")
    logger.info("=" * 70)

    # Load datasets
    logger.info("Loading datasets...")
    X_original, y_original = load_spambase_original()
    X_kstar, y_kstar = load_spambase_kstar_real()

    logger.info(f"  Original dataset: X={X_original.shape}, classes={np.unique(y_original)}")
    logger.info(f"    Class distribution: {np.bincount(y_original)}")
    logger.info(f"  K*-clustering dataset: X={X_kstar.shape}, clusters={np.unique(y_kstar)}")
    logger.info(f"    Cluster distribution: {np.bincount(y_kstar)}")

    # Define datasets
    datasets = {
        "original_2_classes": (X_original, y_original),
        "kstar_classes": (X_kstar, y_kstar),
    }

    # Define classifiers with their param grids
    base_classifiers_config = [
        ("GaussianBayes", GaussianBayesClassifier()),
        ("BayesianKNN", BayesianKNNClassifier()),
        ("ParzenWindow", ParzenWindowClassifier()),
        ("LogisticRegression", CustomLogisticRegression()),
    ]

    # Create MajorityVoting ensemble from the first 4 classifiers
    voting_estimators = [
        (name, clf.__class__()) for name, clf in base_classifiers_config
    ]
    majority_voting = MajorityVotingClassifier(estimators=voting_estimators)

    classifiers_config = base_classifiers_config + [("MajorityVoting", majority_voting)]

    # Get CV splitters
    outer_cv, inner_cv = get_nested_cv_splitters()

    # Collect results
    results = []
    total_iterations = 0
    completed_iterations = 0

    # Pre-calculate total for progress tracking
    n_folds = sum(1 for _ in outer_cv.split(X_original, y_original))
    n_datasets = len(datasets)
    n_classifiers = len(classifiers_config)
    total_expected = n_folds * n_datasets * n_classifiers

    logger.info(f"Total iterations expected: {total_expected}")
    logger.info(f"  ({n_folds} outer folds × {n_datasets} datasets × {n_classifiers} classifiers)")

    # Iterate over datasets
    for dataset_name, (X, y) in datasets.items():
        logger.info(f"\nProcessing dataset: {dataset_name} (n_jobs={n_jobs})")
        outer_splits = list(outer_cv.split(X, y))

        if n_jobs == 1:
            for fold_idx, (train_idx, test_idx) in enumerate(outer_splits):
                repetition = fold_idx // 10
                fold_in_rep = fold_idx % 10
                if verbose >= 1 and fold_idx % 10 == 0:
                    logger.info(
                        f"  Repetition {repetition + 1}/30, fold {fold_in_rep + 1}/10..."
                    )
                fold_rows = _evaluate_outer_fold(
                    dataset_name,
                    fold_idx,
                    train_idx,
                    test_idx,
                    X,
                    y,
                    inner_cv,
                    classifiers_config,
                    base_classifiers_config,
                )
                results.extend(fold_rows)
                completed_iterations += len(fold_rows)
        else:
            fold_batches = Parallel(n_jobs=n_jobs, verbose=10 if verbose >= 1 else 0)(
                delayed(_evaluate_outer_fold)(
                    dataset_name,
                    fold_idx,
                    train_idx,
                    test_idx,
                    X,
                    y,
                    inner_cv,
                    classifiers_config,
                    base_classifiers_config,
                )
                for fold_idx, (train_idx, test_idx) in enumerate(outer_splits)
            )
            for fold_rows in fold_batches:
                results.extend(fold_rows)
                completed_iterations += len(fold_rows)

    logger.info(f"\nCompleted: {completed_iterations}/{total_expected} iterations")

    # Convert to DataFrame
    df_results = pd.DataFrame(results)

    # Validate shape
    assert len(df_results) == total_expected, \
        f"Expected {total_expected} results, got {len(df_results)}"

    logger.info(f"Results shape: {df_results.shape}")

    # Save to CSV
    output_file = output_dir / "cv_30x10_results.csv"
    df_results.to_csv(output_file, index=False)
    logger.info(f"Results saved to: {output_file}")

    # Print summary statistics
    logger.info("\n" + "=" * 70)
    logger.info("SUMMARY STATISTICS (by dataset and classifier)")
    logger.info("=" * 70)

    for dataset_name in datasets.keys():
        logger.info(f"\nDataset: {dataset_name}")
        df_subset = df_results[df_results["dataset"] == dataset_name]

        for clf_name in [name for name, _ in classifiers_config]:
            df_clf = df_subset[df_subset["classifier"] == clf_name]
            mean_f1 = df_clf["f1"].mean()
            std_f1 = df_clf["f1"].std()
            mean_error = df_clf["error"].mean()
            std_error = df_clf["error"].std()

            logger.info(
                f"  {clf_name:20s} | "
                f"F1={mean_f1:.4f}±{std_f1:.4f} | "
                f"Error={mean_error:.4f}±{std_error:.4f}"
            )

    logger.info("\n" + "=" * 70)
    logger.info("Experiment completed successfully!")
    logger.info("=" * 70)

    return df_results


def main():
    """Parse CLI arguments and run the Question 2 nested CV experiment.

    Returns:
        pd.DataFrame: Results DataFrame returned by ``run_nested_cv_experiment``.
    """
    import argparse

    parser = argparse.ArgumentParser(

        "--output-dir",

    parser.add_argument(
        "--verbose",

        "--resume",
        action="store_true",

    )

    return df_results


if __name__ == "__main__":

## Decisões de Arquitetura: Análise Estatística (Friedman e Nemenyi)

- **Fluxo Condicional (Omnibus e Post-hoc):** A função `perform_friedman_nemenyi` consolida o teste de Friedman e o pós-teste de Nemenyi. O *post-hoc* só é executado se o teste *omnibus* rejeitar a hipótese nula global de equivalência entre os modelos ($p < \alpha$, padrão $\alpha = 0.05$), o que evita comparações múltiplas sem respaldo global. Como o Nemenyi é mais conservador que o Friedman, é possível que o Friedman rejeite $H_0$ e nenhum par apareça como significativo na matriz de p-valores. Nesse caso, o resultado é reportado como tal: há diferença global, mas ela não é atribuível a um par específico.

- **Definição dos blocos:** Cada bloco do Friedman é um par `(repetição, fold)`. Com $30 \times 10$-folds, temos $N = 300$ blocos, e cada linha do `DataFrame` de entrada corresponde a um deles. A ordem das linhas deve ser idêntica para todos os classificadores (recomenda-se um `MultiIndex` `(repetição, fold)` para garantir o alinhamento).

- **Limitação: blocos não independentes:** Os 300 blocos **não são independentes**, pois os conjuntos de treino se sobrepõem entre folds e repetições. O teste de Friedman assume blocos independentes, e, conforme Demšar (2006, §3.2.3), não há teste conhecido que corrija essa dependência. Consequência: os p-valores tendem a ser **otimistas**, e as conclusões valem para esta base de dados, não generalizam para outros domínios.

- **Interpretação com $N = 300$ e $k = 5$:** A Diferença Crítica (CD) do Nemenyi é $CD = q_\alpha \sqrt{\frac{k(k+1)}{6N}} \approx 0{,}352$ de rank médio. Por ser tão pequena, quase qualquer diferença será estatisticamente significativa. Por isso, o relatório apresenta, junto aos p-valores, a **diferença média da métrica com seu intervalo de confiança**, para distinguir significância estatística de relevância prática.

- **Abstração por Métrica:** A função é agnóstica à métrica (Taxa de Erro, Precisão, Cobertura, F-measure). Ela recebe um `pandas.DataFrame` com shape `(n_blocos, n_classificadores)`, em que cada coluna é um modelo e cada linha é um bloco, e o nome da métrica é usado apenas no log. Isso evita duplicação de código.

- **Validação de Entrada (*fail-fast*):** A função rejeita entradas que não sejam `DataFrame` (`TypeError`), que contenham NaN/infinito, que estejam vazias, que tenham menos de dois blocos, colunas repetidas ou `alpha` fora de `(0, 1)`. O Friedman exige ao menos 3 classificadores.

- **Integração Scipy e Scikit-Posthocs:** A estatística de Friedman é delegada a `scipy.stats.friedmanchisquare` e o Nemenyi a `scikit-posthocs` (`posthoc_nemenyi_friedman`), que devolve a matriz de p-valores par a par. Quando o Friedman não rejeita $H_0$, o campo `nemenyi_p_values` do resultado é `None`.

- **Retorno:** `perform_friedman_nemenyi` retorna `FriedmanNemenyiResult` com a estatística, p-valor e matriz de Nemenyi; a matriz é `None` quando o teste global não é significativo.

- **Execução Q2:** Rode `python -m src.analysis.statistical_analysis` para calcular ICs de 95% usando as médias das 30 repetições, imprimir métricas e testes no terminal, e salvar os arquivos descritos em [Q2_STATISTICAL_ANALYSIS.md](../../analysis/Q2_STATISTICAL_ANALYSIS.md). Para o Friedman, os 300 folds externos continuam sendo os blocos alinhados, portanto a dependência entre folds deve ser considerada na interpretação.

import json
import numpy as np
import pandas as pd
from pathlib import Path
from sklearn.model_selection import RepeatedStratifiedKFold, StratifiedShuffleSplit, StratifiedKFold

def load_spambase_original(filepath="dataset/spambase.data"):
    """Load the original Spambase dataset with its two a priori classes.

    Args:
        filepath (str or Path, optional): Path to the Spambase data file.
            Defaults to "dataset/spambase.data".

    Returns:
        tuple[np.ndarray, np.ndarray]: A tuple ``(X, y)`` containing the feature
            matrix of shape ``(n_samples, 57)`` and the original binary class
            labels (1 = spam, 0 = non-spam).
    """
    # The data file has no header, so read it without column names.
    df = pd.read_csv(filepath, header=None)

    # The first 57 columns contain the features.
    X = df.iloc[:, :-1].values
    # The last column contains the target labels.
    y = df.iloc[:, -1].values

    return X, y

def load_spambase_kstar_mock(filepath="dataset/spambase.data", k_star=3, random_state=42):
    """Load the Spambase features with mock labels for ``K*`` classes.

    This is a temporary mock implementation until the Double k-means
    integration is complete.

    Args:
        filepath (str or Path, optional): Path to the Spambase data file.
            Defaults to "dataset/spambase.data".
        k_star (int, optional): Number of classes to generate. Defaults to 3.
        random_state (int, optional): Seed used to generate reproducible mock
            labels. Defaults to 42.

    Returns:
        tuple[np.ndarray, np.ndarray]: A tuple ``(X, y_mock)`` containing the
            feature matrix and randomly generated integer labels.
    """
    X, _ = load_spambase_original(filepath)

    # Generate random labels to simulate Double k-means output.
    rng = np.random.default_rng(random_state)
    y_mock = rng.integers(0, k_star, size=X.shape[0])

    return X, y_mock

def load_spambase_kstar_real(filepath="dataset/spambase.data",
                             q1_labels_path="results/q1_object_labels.csv",
                             q1_selection_path="results/q1_selection.json"):
    """Load Spambase features with K*-clustering labels from Q1 analysis.

    This function loads the real K*-clustering partition discovered by Double k-means
    in Q1, instead of using mock random labels. The labels are read from the CSV file
    saved during Q1 evaluation, ensuring exact reproducibility.

    Args:
        filepath (str or Path, optional): Path to the Spambase data file.
            Defaults to "dataset/spambase.data".
        q1_labels_path (str or Path, optional): Path to ``q1_object_labels.csv``
            containing ``row_index``, ``original_class``, and ``cluster``.
            Defaults to "results/q1_object_labels.csv".
        q1_selection_path (str or Path, optional): Path to ``q1_selection.json``
            containing metadata including ``K*`` and ``H*``. Defaults to
            "results/q1_selection.json".

    Returns:
        tuple[np.ndarray, np.ndarray]: A tuple ``(X, y_kstar)`` where ``X`` is
            the feature matrix of shape ``(n_samples, 57)`` and ``y_kstar`` is
            the vector of cluster labels from the winning Q1 partition.

    Raises:
        FileNotFoundError: If ``q1_labels_path`` or ``q1_selection_path`` does
            not exist.
        ValueError: If labels cannot be loaded or have an inconsistent shape or
            cluster count.
    """
    # Load features from original dataset.
    X, _ = load_spambase_original(filepath)

    # Verify that q1_labels and q1_selection files exist.
    q1_labels_file = Path(q1_labels_path)
    q1_selection_file = Path(q1_selection_path)

    if not q1_labels_file.exists():
        raise FileNotFoundError(
            f"Q1 labels file not found: {q1_labels_path}\n"
            f"Run Q1 analysis first: python -m src.analysis.clustering_evaluation"
        )

    if not q1_selection_file.exists():
        raise FileNotFoundError(
            f"Q1 selection metadata not found: {q1_selection_path}\n"
            f"Run Q1 analysis first: python -m src.analysis.clustering_evaluation"
        )

    # Load Q1 selection metadata to verify K* and H*.
    with open(q1_selection_file, 'r') as f:
        q1_metadata = json.load(f)

    k_star = q1_metadata["K"]
    h_star = q1_metadata["H"]

    # Load cluster labels from Q1 CSV (preserves row order).
    df_labels = pd.read_csv(q1_labels_file)

    if df_labels.shape[0] != X.shape[0]:
        raise ValueError(
            f"Label count mismatch: CSV has {df_labels.shape[0]} rows, "
            f"but dataset has {X.shape[0]} rows. Check data integrity."
        )

    # Extract cluster column and verify it contains only valid cluster indices.
    y_kstar = df_labels["cluster"].values
    n_unique_clusters = len(np.unique(y_kstar))

    if n_unique_clusters != k_star:
        raise ValueError(
            f"Cluster count mismatch: expected {k_star} clusters (from Q1 metadata), "
            f"but found {n_unique_clusters} in the labels file."
        )

    return X, y_kstar

def get_nested_cv_splitters():
    """Create splitters for the nested cross-validation procedure.

    Returns:
        tuple[RepeatedStratifiedKFold, StratifiedKFold]: A tuple
            ``(outer_cv, inner_cv)`` where ``outer_cv`` is a
            ``RepeatedStratifiedKFold`` (10 folds, 30 repetitions) for outer
            evaluation and ``inner_cv`` is a ``StratifiedKFold`` (5 folds) for
            inner hyperparameter tuning.
    """
    outer_cv = RepeatedStratifiedKFold(n_splits=10, n_repeats=30, random_state=42)
    inner_cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
    return outer_cv, inner_cv

def get_learning_curve_splits(X, y, n_repeats=10, random_state=42):
    """Generate stratified train/test splits for a learning curve.

    Args:
        X (np.ndarray): Feature matrix used to generate the splits.
        y (np.ndarray): Target labels used for stratification.
        n_repeats (int, optional): Number of randomized splits per training
            size to allow variance calculation. Defaults to 10.
        random_state (int, optional): Seed used to generate reproducible random
            splits. Defaults to 42.

    Returns:
        dict[float, list[tuple[np.ndarray, np.ndarray]]]: Dictionary mapping
            each training-set proportion from 0.05 to 0.95 to a list of
            ``(train_idx, test_idx)`` index tuples.
    """
    splits = {}

    # Vary the training-set proportion from 5% to 95% in 5% increments.
    train_sizes = np.arange(0.05, 0.96, 0.05)

    for train_size in train_sizes:
        train_size_rounded = round(train_size, 2)

        # StratifiedShuffleSplit preserves the class distribution in each split.
        # n_splits is set to n_repeats to generate multiple independent samples.
        splitter = StratifiedShuffleSplit(
            n_splits=n_repeats, 
            train_size=train_size_rounded, 
            random_state=random_state
        )

        # Store all generated splits for this specific training size as a list
        splits[train_size_rounded] = list(splitter.split(X, y))

    return splits

"""Statistical tests for comparing classifiers across aligned CV folds."""

from dataclasses import dataclass

import numpy as np
import pandas as pd
import scikit_posthocs as sp
from scipy.stats import friedmanchisquare


@dataclass(frozen=True)
class FriedmanNemenyiResult:
    """Results of the Friedman test and its conditional Nemenyi post-hoc test.

    Args:
        statistic (float): Friedman chi-square test statistic.
        p_value (float): Two-sided p-value associated with the Friedman test.
        nemenyi_p_values (pd.DataFrame or None): Pairwise Nemenyi p-value matrix
            when ``p_value < alpha``, or ``None`` otherwise.
    """

    statistic: float
    p_value: float
    nemenyi_p_values: pd.DataFrame | None


def perform_friedman_nemenyi(
    df_results: pd.DataFrame,
    metric_name: str,
    alpha: float = 0.05,
) -> FriedmanNemenyiResult:
    """Compare classifiers on aligned blocks, applying Nemenyi when warranted.

    Rows must represent the same folds for every classifier; columns represent
    classifiers. The post-hoc test is run only when the omnibus test is
    significant at ``alpha``.

    Args:
        df_results (pd.DataFrame): DataFrame of shape (n_blocks, n_classifiers)
            containing aligned fold scores for each classifier.
        metric_name (str): Display name of the evaluated metric.
        alpha (float, optional): Significance threshold in (0, 1). Defaults to
            0.05.

    Returns:
        FriedmanNemenyiResult: Dataclass containing the Friedman statistic,
            p-value, and optional Nemenyi pairwise p-value DataFrame.

    Raises:
        TypeError: If ``df_results`` is not a DataFrame or contains non-numeric
            entries.
        ValueError: If ``df_results`` is empty, has fewer than 2 blocks or 3
            classifiers, has duplicate columns, contains NaN/inf values, has
            identical values across all classifiers, or if ``alpha`` is outside
            (0, 1).
    """
    if not isinstance(df_results, pd.DataFrame):
        raise TypeError("Results must be a pandas DataFrame with aligned folds.")
    if df_results.empty:
        raise ValueError("The results DataFrame is empty.")
    if len(df_results) < 2:
        raise ValueError("At least two aligned blocks are required.")
    if len(df_results.columns) < 3:
        raise ValueError("The Friedman test requires at least three classifiers.")
    if df_results.columns.has_duplicates:
        raise ValueError("Classifier names must be unique.")
    if not 0 < alpha < 1:
        raise ValueError("alpha must be between 0 and 1.")
    if df_results.isna().any().any():
        raise ValueError("Results contain missing values.")
    if not df_results.map(lambda value: pd.api.types.is_number(value)).all().all():
        raise TypeError("All classifier results must be numeric.")
    if not df_results.map(lambda value: pd.notna(value) and abs(value) != float("inf")).all().all():
        raise ValueError("Results must contain only finite values.")
    first_classifier = df_results.iloc[:, 0].to_numpy()
    if all(
        np.array_equal(first_classifier, df_results[classifier].to_numpy())
        for classifier in df_results.columns[1:]
    ):
        raise ValueError("The Friedman test is undefined when all results are identical.")

    statistic, p_value = friedmanchisquare(
        *(df_results[classifier].to_numpy() for classifier in df_results.columns),
        nan_policy="raise",
    )
    if not np.isfinite(statistic) or not np.isfinite(p_value):
        raise ValueError("The Friedman test is undefined for these classifier results.")
    nemenyi_p_values = (
        sp.posthoc_nemenyi_friedman(df_results)
        if p_value < alpha
        else None
    )

    print(f"\n{metric_name.upper()}: Friedman chi-square={statistic:.4f}, p={p_value:.4e}")
    if nemenyi_p_values is None:
        print(f"  No significant global difference at alpha={alpha:.3f}; Nemenyi not run.")
    else:
        print(f"  Significant global difference at alpha={alpha:.3f}; Nemenyi p-values:")
        print(nemenyi_p_values.to_string(float_format=lambda value: f"{value:.4f}"))

    return FriedmanNemenyiResult(
        statistic=float(statistic),
        p_value=float(p_value),
        nemenyi_p_values=nemenyi_p_values,
    )

def _save_figure(figure, path):
    """Atomically write a Matplotlib figure to disk as a PNG file.

    Args:
        figure (Figure): Matplotlib figure instance to render.
        path (Path): Destination file path.
    """
    try:
        with _atomic_output(path) as stream:
            figure.savefig(stream, format="png", dpi=160)
    finally:
        figure.clear()


def plot_q1_results(best, contingency, plots_dir):
    """Save contingency, block prototypes, and objective convergence plots.

    Args:
        best (dict): Winning Double k-means run dictionary containing ``K``,
            ``H``, ``run``, ``prototypes``, and ``objective_history``.
        contingency (pd.DataFrame): Contingency table comparing original classes
            and object clusters.
        plots_dir (str or Path): Output directory where PNG plots are written.
    """
    plots_dir = Path(plots_dir)
    plots_dir.mkdir(parents=True, exist_ok=True)

    figure = Figure(figsize=(7, 4.5), layout="constrained")
    axes = figure.subplots()
    counts = contingency.to_numpy()
    image = axes.imshow(counts, cmap="Blues", aspect="auto")
    axes.set_xticks(range(len(contingency.columns)), labels=contingency.columns)
    axes.set_yticks(range(len(contingency.index)), labels=contingency.index)
    axes.set_xlabel("Object cluster (arbitrary ID)")
    axes.set_ylabel("Original class")
    axes.set_title("Q1: original classes vs object clusters")
    for (i, j), count in np.ndenumerate(counts):
        axes.text(j, i, str(count), ha="center", va="center",
                  color="white" if count > counts.max() / 2 else "black")
    figure.colorbar(image, ax=axes, label="Number of observations")
    _save_figure(figure, plots_dir / "confusion_matrix_q1.png")

    figure = Figure(figsize=(7, 4.5), layout="constrained")
    axes = figure.subplots()
    G = best["prototypes"]
    limit = max(float(np.max(np.abs(G))), np.finfo(float).eps)
    image = axes.imshow(G, cmap="RdBu_r", vmin=-limit, vmax=limit, aspect="auto")
    axes.set_xticks(range(best["H"]))
    axes.set_yticks(range(best["K"]))
    axes.set_xlabel("Variable cluster")
    axes.set_ylabel("Object cluster")
    axes.set_title(f"Block prototypes G (K={best['K']}, H={best['H']})")
    for (i, j), value in np.ndenumerate(G):
        axes.text(j, i, f"{value:.4f}", ha="center", va="center",
                  color="white" if abs(value) > limit / 2 else "black")
    figure.colorbar(image, ax=axes, label="Mean in standardized feature units")
    _save_figure(figure, plots_dir / "prototypes_heatmap.png")

    figure = Figure(figsize=(7, 4.5), layout="constrained")
    axes = figure.subplots()
    values = best["objective_history"]
    axes.plot(np.arange(len(values)), values, marker="o", markersize=3)
    axes.set_xlabel("Iteration (0 = initialization)")
    axes.set_ylabel("Objective W(G, U, V)")
    axes.set_title(f"Q1 convergence (K={best['K']}, H={best['H']}, seed={best['run']})")
    axes.grid(alpha=0.25)
    _save_figure(figure, plots_dir / "objective_convergence.png")

import numpy as np
from sklearn.model_selection import train_test_split, GridSearchCV
from src.utils.dataloader import load_spambase_original

def test_bayesian_knn():
    """Test the BayesianKNNClassifier class with the Spambase dataset."""
    print("=== TESTANDO CLASSIFICADOR BAYESIANO K-NN ===")
    print("[INFO] A carregar dataset spambase...")
    
    # Carrega os dados reais do projeto
    X, y = load_spambase_original("dataset/spambase.data")
    
    # Divisão 70/30 estratificada apenas para o teste isolado
    X_train, X_test, y_train, y_test = train_test_split(
        X, y, test_size=0.3, stratify=y, random_state=42
    )

    print("\n--- 1. Teste de Fit e Predict Direto ---")
    clf = BayesianKNNClassifier(n_neighbors=5, metric='euclidean')
    clf.fit(X_train, y_train)
    
    preds = clf.predict(X_test)
    acuracia = np.mean(preds == y_test)
    print(f"[OK] Acurácia inicial (k=5, euclidean): {acuracia:.4f}")
    print(f"     Classes encontradas pelo modelo: {clf.classes_}")

    print("\n--- 2. Teste de Ajuste de Hiperparâmetros (CV Interna) ---")
    # Puxa a grelha diretamente do método estático da nossa classe
    param_grid = BayesianKNNClassifier.get_param_grid()
    
    # Configura o GridSearchCV simulando a etapa interna (5-folds)
    grid = GridSearchCV(
        BayesianKNNClassifier(), 
        param_grid, 
        cv=5, 
        scoring='f1_macro', 
        n_jobs=-1
    )
    
    print("[INFO] A iniciar busca em grade (5-folds)...")
    grid.fit(X_train, y_train)
    
    print(f"[OK] Busca finalizada!")
    print(f"     Melhores hiperparâmetros: {grid.best_params_}")
    print(f"     Melhor F1-macro média na validação interna: {grid.best_score_:.4f}")
    
    # Testa as predições com o melhor modelo encontrado
    best_preds = grid.predict(X_test)
    best_acc = np.mean(best_preds == y_test)
    print(f"[OK] Acurácia no teste com o melhor modelo: {best_acc:.4f}")

if __name__ == "__main__":
    test_bayesian_knn()

    data, results, plots = experiment
    report = tmp_path / "report" / "Q1_ANALYSIS.md"

    np.testing.assert_array_equal(labels.cluster, best["labels"])

    columns = pd.read_csv(results / "q1_variable_labels.csv")

N_SAMPLES, N_FEATURES = 60, 12
BLOCK_NOISE = 0.5


@pytest.fixture
def random_data():
    """Standard normal data with no group structure."""
    return np.random.RandomState(0).normal(size=(N_SAMPLES, N_FEATURES))


@pytest.fixture
def block_data():
    """Data with 3 object groups and 2 variable groups, and the true labels."""
    rng = np.random.RandomState(1)
    true_rows = rng.permutation(np.repeat(np.arange(3), N_SAMPLES // 3))
    true_cols = rng.permutation(np.repeat(np.arange(2), N_FEATURES // 2))
    block_means = np.array([[0.0, 5.0], [5.0, 0.0], [10.0, 10.0]])
    X = (block_means[np.ix_(true_rows, true_cols)]
         + BLOCK_NOISE * rng.normal(size=(N_SAMPLES, N_FEATURES)))
    return X, true_rows, true_cols


def reference_prototypes(X, rows, cols, K, H):
    """Block means computed with plain loops."""
    G = np.empty((K, H))
    for k in range(K):
        for h in range(H):
            G[k, h] = X[np.ix_(rows == k, cols == h)].mean()
    return G


def reference_objective(X, G, rows, cols):
    """W(G, U, V) computed with plain loops."""
    return sum((X[i, j] - G[rows[i], cols[j]]) ** 2
               for i in range(X.shape[0]) for j in range(X.shape[1]))

# Stopping rule of the slides.

def test_converged_partition_is_a_fixed_point(random_data):
    """After convergence, steps 2 and 3 would not move any object or variable."""
    X, K, H = random_data, 3, 2
    model = DoubleKMeans(K, H, random_state=0).fit(X)
    rows, cols, G = model.row_labels_, model.column_labels_, model.prototypes_
    assert model.converged_

    row_cost = np.array([[np.sum((X[i] - G[k, cols]) ** 2) for k in range(K)]
                         for i in range(N_SAMPLES)])
    col_cost = np.array([[np.sum((X[:, j] - G[rows, h]) ** 2) for h in range(H)]
                         for j in range(N_FEATURES)])

    # Margin of _assign (1e-9) plus room for rounding in these costs.
    for cost, labels in ((row_cost, rows), (col_cost, cols)):
        gain = cost[np.arange(len(labels)), labels] - cost.min(axis=1)
        assert np.all(gain <= 1e-8)

    cost = np.array([[1.0, 1.0 - gain]])
    new_labels = DoubleKMeans._assign(cost, np.array([0]))
    assert new_labels[0] == (1 if moves else 0)


def test_relocation_fills_empty_object_group(random_data):
    X = random_data
    rows = np.arange(N_SAMPLES) % 2  # group 2 of 3 is empty
    cols = np.arange(N_FEATURES) % 2
    G = np.vstack([reference_prototypes(X, rows, cols, 2, 2), np.zeros(2)])

    new_rows, new_G, n_moved = DoubleKMeans._relocate_empty_groups(
        X, G, rows, cols)

    assert n_moved == 1
    assert np.all(np.bincount(new_rows, minlength=3) >= 1)
    assert (reference_objective(X, new_G, new_rows, cols)
            <= reference_objective(X, G, rows, cols))


def test_relocation_fills_empty_variable_group(random_data):
    X = random_data
    rows = np.arange(N_SAMPLES) % 2
    cols = np.arange(N_FEATURES) % 2  # group 2 of 3 is empty
    G = np.hstack([reference_prototypes(X, rows, cols, 2, 2), np.zeros((2, 1))])

    new_cols, new_G_t, n_moved = DoubleKMeans._relocate_empty_groups(
        X.T, G.T, cols, rows)

    assert n_moved == 1
    assert np.all(np.bincount(new_cols, minlength=3) >= 1)
    assert (reference_objective(X, new_G_t.T, rows, new_cols)
            <= reference_objective(X, G, rows, cols))

def test_covariances_are_unregularized_maximum_likelihood_estimates():
    X = np.array([
        [0.0, 0.0],
        [1.0, 0.2],
        [0.2, 1.0],
        [3.0, 3.0],
        [4.0, 3.2],
        [3.2, 4.0],
    ])
    y = np.array([0, 0, 0, 1, 1, 1])

    classifier = GaussianBayesClassifier().fit(X, y)

    for index, label in enumerate(classifier.classes_):
        class_samples = X[y == label]
        centered = class_samples - class_samples.mean(axis=0)
        expected = centered.T @ centered / len(class_samples)
        np.testing.assert_allclose(classifier.covariances_[index], expected)

    probabilities = classifier.predict_proba(X)
    assert np.isfinite(probabilities).all()
    np.testing.assert_allclose(probabilities.sum(axis=1), 1.0)
    assert GaussianBayesClassifier.get_param_grid() == {}

import numpy as np, pandas as pd

def evaluate_performance(y_true, y_pred, subset_name):
    """Calcula e exibe métricas de avaliação do Scikit-Learn."""
    acc = accuracy_score(y_true, y_pred)
    # average='macro' garante o cálculo correto mesmo quando a base tiver K* classes
    prec = precision_score(y_true, y_pred, average='macro', zero_division=0)
    rec = recall_score(y_true, y_pred, average='macro', zero_division=0)
    f1 = f1_score(y_true, y_pred, average='macro', zero_division=0)
    
    print(f"--- Métricas no conjunto de {subset_name} ---")
    print(f"Acurácia : {acc:.4f}")
    print(f"Precisão : {prec:.4f}")
    print(f"Recall   : {rec:.4f}")
    print(f"F1-Score : {f1:.4f}\n")


def test_majority_voting():
    """Test the MajorityVotingClassifier with optimized base models."""
    print("=== TESTANDO CLASSIFICADOR DE VOTO MAJORITÁRIO ===")
    print("[INFO] A carregar dataset spambase...")
    
    X, y = load_spambase_original("dataset/spambase.data")
    
    # Divisão 70/30 estratificada para o teste rápido
    X_train, X_test, y_train, y_test = train_test_split(
        X, y, test_size=0.3, stratify=y, random_state=42
    )

    print("[INFO] A otimizar hiperparâmetros dos classificadores base (GridSearch CV=3)...")
    
    # 1. Bayes Gaussiano
    grid_gauss = GridSearchCV(GaussianBayesClassifier(), GaussianBayesClassifier.get_param_grid(), cv=3, scoring='f1_macro', n_jobs=-1)
    grid_gauss.fit(X_train, y_train)
    
    # 2. Bayes KNN
    grid_knn = GridSearchCV(BayesianKNNClassifier(), BayesianKNNClassifier.get_param_grid(), cv=3, scoring='f1_macro', n_jobs=-1)
    grid_knn.fit(X_train, y_train)
    
    # 3. Parzen Window
    grid_parzen = GridSearchCV(ParzenWindowClassifier(), ParzenWindowClassifier.get_param_grid(), cv=3, scoring='f1_macro', n_jobs=-1)
    grid_parzen.fit(X_train, y_train)
    
    # 4. Regressão Logística
    grid_logreg = GridSearchCV(CustomLogisticRegression(), CustomLogisticRegression.get_param_grid(), cv=3, scoring='f1_macro', n_jobs=-1)
    grid_logreg.fit(X_train, y_train)

    print("[INFO] Melhores hiperparâmetros encontrados. A instanciar o Voto Majoritário...")
    # Estrutura exigida pelo VotingClassifier com os estimadores já otimizados
    estimators = [
        ('gaussian', grid_gauss.best_estimator_),
        ('knn', grid_knn.best_estimator_),
        ('parzen', grid_parzen.best_estimator_),
        ('log_reg', grid_logreg.best_estimator_)
    ]

    ensemble = MajorityVotingClassifier(estimators=estimators)
    ensemble.fit(X_train, y_train)


    # 1) A CV interna está otimista em relação ao teste?
    for name, grid in [('gaussian', grid_gauss), ('knn', grid_knn),
                    ('parzen', grid_parzen), ('log_reg', grid_logreg)]:
        print(f"{name:9s} F1 CV interna = {grid.best_score_:.4f}")

    # 2) O k=1 ganha por muito ou por pouco?
    res = pd.DataFrame(grid_knn.cv_results_)[['param_metric', 'param_n_neighbors', 'mean_test_score']]
    print(res.sort_values('mean_test_score', ascending=False).head(10))

    # 3) Duplicatas favorecem k=1 e h pequeno
    print("Linhas duplicadas:", pd.DataFrame(X).duplicated().sum())

    # 4) O wrapper está votando corretamente? (empate 2-2 -> classe 0)
    votes = np.column_stack([est.predict(X_test) for _, est in estimators])
    manual = (votes.sum(axis=1) > 2).astype(int)
    print("Voto manual == ensemble:", np.array_equal(manual, ensemble.predict(X_test)))
    tie = votes.sum(axis=1) == 2
    print("Fração de spam real entre os empates:", (np.asarray(y_test)[tie] == 1).mean())

    votes = np.column_stack([est.predict(X_test) for _, est in estimators])
    n_ties = np.sum(votes.sum(axis=1) == 2)  # binário: 2 votos na classe 1 = empate 2-2
    print(f"Empates 2-2 no teste: {n_ties} de {len(y_test)} ({n_ties/len(y_test):.1%})")
    
    print(f"[OK] Classes mapeadas pelo ensemble: {ensemble.classes_}\n")

    print("[INFO] A gerar predições e avaliar desempenho...")
    preds_train = ensemble.predict(X_train)
    preds_test = ensemble.predict(X_test)
    
    # Avaliação completa de métricas em Treino e Teste
    evaluate_performance(y_train, preds_train, "Treino")
    evaluate_performance(y_test, preds_test, "Teste")

def test_resume_reuses_completed_pairs_and_repairs_outputs(spambase_file, tmp_path,
                                                         monkeypatch):
    output_dir = tmp_path / "resumed"
    original_fit = question_1.DoubleKMeans.fit
    calls = []

    def interrupt_fit(model, X, y=None):
        calls.append((model.n_row_clusters, model.n_col_clusters, model.random_state))
        if len(calls) == 4:
            raise RuntimeError("interrupted")
        return original_fit(model, X, y)

    monkeypatch.setattr(question_1.DoubleKMeans, "fit", interrupt_fit)
    with pytest.raises(RuntimeError, match="interrupted"):
        question_1.run_question_1(spambase_file, output_dir, n_runs=2)
    assert len(load_history(output_dir)["runs"]) == 2
    (output_dir / "q1_executions_summary.csv").unlink()
    (output_dir / "q1_best_dkm_K2_H1.npz").unlink()

    calls.clear()

    def tracking_fit(model, X, y=None):
        calls.append((model.n_row_clusters, model.n_col_clusters, model.random_state))
        return original_fit(model, X, y)

    monkeypatch.setattr(question_1.DoubleKMeans, "fit", tracking_fit)
    resumed = question_1.run_question_1(spambase_file, output_dir, n_runs=2, resume=True)
    assert len(calls) == 16
    assert all((K, H) != (2, 1) for K, H, _ in calls)
    uninterrupted_dir = tmp_path / "uninterrupted"
    uninterrupted = question_1.run_question_1(spambase_file, uninterrupted_dir, n_runs=2)
    pd.testing.assert_frame_equal(resumed, uninterrupted)
    for actual, expected in zip(load_history(output_dir)["runs"],
                                load_history(uninterrupted_dir)["runs"]):
        for key in expected:
            np.testing.assert_array_equal(actual[key], expected[key])

    calls.clear()
    question_1.run_question_1(spambase_file, output_dir, n_runs=2, resume=True)
    assert calls == []

    def test_run_nested_cv_returns_dataframe(self, fast_q2_pipeline_env):
        """Verify that run_nested_cv_experiment returns a pandas DataFrame."""
        df, _ = fast_q2_pipeline_env
        assert isinstance(df, pd.DataFrame), f"Expected DataFrame, got {type(df)}"

    def test_output_csv_is_created(self, fast_q2_pipeline_env):
        """Verify that the output CSV file is created."""
        _, tmpdir = fast_q2_pipeline_env
        output_file = tmpdir / "cv_30x10_results.csv"
        assert output_file.exists(), f"Output CSV not found: {output_file}"

    def test_results_dataframe_shape(self, fast_q2_pipeline_env):
        """Verify results have exactly 3000 rows (300 folds × 2 datasets × 5 classifiers)."""
        df, _ = fast_q2_pipeline_env
        expected_rows = 300 * 2 * 5
        assert len(df) == expected_rows, \
            f"Expected {expected_rows} rows, got {len(df)}"

    def test_results_have_required_columns(self, fast_q2_pipeline_env):
        """Verify all required columns are present in results."""
        df, _ = fast_q2_pipeline_env
        required_columns = {
            "fold_id", "repetition", "fold_in_rep", "dataset", "classifier",
            "error", "precision", "recall", "f1", "best_params",
            "n_train", "n_test", "best_f1_inner"
        }
        actual_columns = set(df.columns)
        assert required_columns.issubset(actual_columns), \
            f"Missing columns: {required_columns - actual_columns}"

    def test_fold_structure_30x10(self, fast_q2_pipeline_env):
        """Verify the 30×10 fold structure is correct."""
        df, _ = fast_q2_pipeline_env
        for dataset in df["dataset"].unique():
            for classifier in df["classifier"].unique():
                df_subset = df[(df["dataset"] == dataset) & (df["classifier"] == classifier)]
                repetitions = df_subset["repetition"].unique()
                assert len(repetitions) == 30, \
                    f"Expected 30 repetitions, got {len(repetitions)}"
                for rep in repetitions:
                    df_rep = df_subset[df_subset["repetition"] == rep]
                    folds_in_rep = df_rep["fold_in_rep"].unique()
                    assert len(folds_in_rep) == 10, \
                        f"Expected 10 folds per repetition, got {len(folds_in_rep)}"

    def test_datasets_coverage(self, fast_q2_pipeline_env):
        """Verify both datasets are present in results."""
        df, _ = fast_q2_pipeline_env
        datasets = set(df["dataset"].unique())
        expected_datasets = {"original_2_classes", "kstar_classes"}
        assert datasets == expected_datasets, \
            f"Expected datasets {expected_datasets}, got {datasets}"

    def test_classifiers_coverage(self, fast_q2_pipeline_env):
        """Verify all 5 classifiers are present in results."""
        df, _ = fast_q2_pipeline_env
        classifiers = set(df["classifier"].unique())
        expected_classifiers = {
            "GaussianBayes", "BayesianKNN", "ParzenWindow",
            "LogisticRegression", "MajorityVoting"
        }
        assert classifiers == expected_classifiers, \
            f"Expected classifiers {expected_classifiers}, got {classifiers}"

    def test_metrics_are_numeric_and_valid(self, fast_q2_pipeline_env):
        """Verify metrics are numeric and within valid ranges."""
        df, _ = fast_q2_pipeline_env

        for metric in ["error", "precision", "recall", "f1", "best_f1_inner"]:
            assert np.issubdtype(df[metric].dtype, np.number), \
                f"Metric {metric} is not numeric"

        assert (df["error"] >= 0).all() and (df["error"] <= 1).all(), \
            "error values outside [0, 1]"
        assert (df["precision"] >= 0).all() and (df["precision"] <= 1).all(), \
            "precision values outside [0, 1]"
        assert (df["recall"] >= 0).all() and (df["recall"] <= 1).all(), \
            "recall values outside [0, 1]"
        assert (df["f1"] >= 0).all() and (df["f1"] <= 1).all(), \
            "f1 values outside [0, 1]"

    def test_no_missing_values_in_metrics(self, fast_q2_pipeline_env):
        """Verify no NaN or inf values in metrics."""
        df, _ = fast_q2_pipeline_env
        for metric in ["error", "precision", "recall", "f1"]:
            assert not df[metric].isna().any(), f"NaN values in {metric}"
            assert not np.isinf(df[metric]).any(), f"Inf values in {metric}"
        voting_scores = df.loc[
            df["classifier"] == "MajorityVoting", "best_f1_inner"
        ]
        assert voting_scores.isna().all(), \
            "MajorityVoting has no single inner-CV tuning score."
        base_scores = df.loc[
            df["classifier"] != "MajorityVoting", "best_f1_inner"
        ]
        assert base_scores.notna().all()

    def test_fold_ids_are_sequential(self, fast_q2_pipeline_env):
        """Verify fold_id values are sequential from 0 to 299."""
        df, _ = fast_q2_pipeline_env
        for dataset in df["dataset"].unique():
            for classifier in df["classifier"].unique():
                df_subset = df[(df["dataset"] == dataset) & (df["classifier"] == classifier)]
                fold_ids = sorted(df_subset["fold_id"].unique())
                assert fold_ids == list(range(300)), \
                    f"fold_ids not sequential for {dataset}/{classifier}"

    def test_train_test_split_sizes(self, fast_q2_pipeline_env):
        """Verify training and test set sizes are consistent."""
        df, _ = fast_q2_pipeline_env
        for fold_id in df["fold_id"].unique():
            df_fold = df[df["fold_id"] == fold_id]
            n_train_values = df_fold["n_train"].unique()
            n_test_values = df_fold["n_test"].unique()
            assert len(n_train_values) == 1, \
                f"Inconsistent n_train for fold_id {fold_id}"
            assert len(n_test_values) == 1, \
                f"Inconsistent n_test for fold_id {fold_id}"

        total_samples = 4601
        for _, row in df.iterrows():
            assert row["n_train"] + row["n_test"] == total_samples, \
                f"Train+test split doesn't equal {total_samples}"

    def test_best_params_not_empty(self, fast_q2_pipeline_env):
        """Verify best_params are recorded for each result."""
        df, _ = fast_q2_pipeline_env
        assert not df["best_params"].isna().any(), \
            "Some best_params are missing"
        assert (df["best_params"].str.len() > 0).all(), \
            "Some best_params are empty strings"
        voting_params = df.loc[
            df["classifier"] == "MajorityVoting", "best_params"
        ].map(ast.literal_eval)
        assert all(
            set(params)
            == {
                "GaussianBayes",
                "BayesianKNN",
                "ParzenWindow",
                "LogisticRegression",
            }
            for params in voting_params
        )

    def test_f1_relates_to_error(self, fast_q2_pipeline_env):
        """Verify logical relationship: higher F1 should correlate with lower error."""
        df, _ = fast_q2_pipeline_env
        correlation = df["f1"].corr(df["error"])
        assert correlation < -0.5, \
            f"F1 and error not strongly negatively correlated: {correlation}"

    def test_outer_fold_structure_consistency(self, fast_q2_pipeline_env):
        """Verify outer fold structure is consistent across datasets and classifiers."""
        df, _ = fast_q2_pipeline_env
        ref_df = df[(df["dataset"] == "original_2_classes") &
                    (df["classifier"] == "GaussianBayes")]
        ref_fold_structure = ref_df[["fold_id", "repetition", "fold_in_rep",
                                      "n_train", "n_test"]].drop_duplicates()

        for dataset in df["dataset"].unique():
            for classifier in df["classifier"].unique():
                df_subset = df[(df["dataset"] == dataset) &
                               (df["classifier"] == classifier)]
                fold_structure = df_subset[["fold_id", "repetition", "fold_in_rep",
                                             "n_train", "n_test"]].drop_duplicates()
                assert len(fold_structure) == len(ref_fold_structure), \
                    f"Inconsistent fold structure for {dataset}/{classifier}"

    def test_csv_file_matches_dataframe(self, fast_q2_pipeline_env):
        """Verify saved CSV file matches returned DataFrame."""
        df_returned, tmpdir = fast_q2_pipeline_env
        df_loaded = pd.read_csv(tmpdir / "cv_30x10_results.csv")
        assert df_returned.shape == df_loaded.shape, \
            "DataFrame shape doesn't match saved CSV"
        pd.testing.assert_frame_equal(df_returned, df_loaded, atol=1e-10)

    def test_experiment_deterministic(self, fast_q2_pipeline_env):
        """Verify experiment is deterministic (same seed gives same results)."""
        df1, _ = fast_q2_pipeline_env
        with tempfile.TemporaryDirectory() as tmpdir2:
            df2 = run_nested_cv_experiment(output_dir=tmpdir2, verbose=0)

        pd.testing.assert_frame_equal(
            df1[["error", "precision", "recall", "f1"]].reset_index(drop=True),
            df2[["error", "precision", "recall", "f1"]].reset_index(drop=True),
            atol=1e-10
        )

    def test_realistic_error_rates(self, fast_q2_pipeline_env):
        """Verify error rates are realistic (not all 0 or 1)."""
        df, _ = fast_q2_pipeline_env
        error_min = df["error"].min()
        error_max = df["error"].max()
        assert 0 <= error_min < error_max <= 1, \
            f"Error rates not realistic: min={error_min}, max={error_max}"


def test_majority_voting_uses_tuned_base_estimators(monkeypatch):
    parameter_grids = (
        (GaussianBayesClassifier, {}),
        (BayesianKNNClassifier, {"n_neighbors": [1], "metric": ["euclidean"]}),
        (ParzenWindowClassifier, {"bandwidth": [1.0]}),
        (CustomLogisticRegression, {"C": [1.0], "solver": ["lbfgs"]}),
    )
    for classifier, grid in parameter_grids:
        monkeypatch.setattr(classifier, "get_param_grid", staticmethod(lambda grid=grid: grid))

    rng = np.random.default_rng(42)
    X = rng.normal(size=(60, 4))
    y = np.repeat([0, 1], 30)
    X[y == 1, 0] += 1
    _, inner_cv = get_nested_cv_splitters()

    voting_classifier, selected_params = _fit_tuned_majority_voting(
        X,
        y,
        inner_cv,
        n_jobs=1,
    )

    assert set(selected_params) == {
        "GaussianBayes",
        "BayesianKNN",
        "ParzenWindow",
        "LogisticRegression",
    }
    assert set(voting_classifier.model_.named_estimators_) == set(selected_params)
    assert voting_classifier.predict(X[:5]).shape == (5,)

import json

import numpy as np

def read_history(results):
    with (results / "q1_full_history.pkl").open("rb") as stream:
        return pickle.load(stream)


def write_history(results, history):
    with (results / "q1_full_history.pkl").open("wb") as stream:
        pickle.dump(history, stream)


def test_full_scores_and_selection_match_reference(experiment, monkeypatch, caplog):
    data, results, plots = experiment
    history = read_history(results)
    original_checkpoint = (results / "q1_full_history.pkl").read_bytes()

    def forbid_refit(*args, **kwargs):
        pytest.fail("Silhouette analysis must not fit clustering or scaler.")

    monkeypatch.setattr(DoubleKMeans, "fit", forbid_refit)
    monkeypatch.setattr(StandardScaler, "fit", forbid_refit)
    X, _ = load_spambase_original(data)
    X_scaled = (X - history["scaler_mean"]) / history["scaler_scale"]
    scores = silhouette.analyze_silhouette(data, results, plots)
    assert list(zip(scores.K, scores.H)) == list(Q1_GRID)
    assert scores.n_runs.eq(2).all()
    for row in scores.itertuples(index=False):
        runs = [r for r in history["runs"] if (r["K"], r["H"]) == (row.K, row.H)]
        best = min(runs, key=lambda r: r["objective"])
        assert row.best_run_id == best["run"]
        assert row.best_objective == best["objective"]
        assert row.silhouette == pytest.approx(
            silhouette_score(X_scaled, best["labels"], metric="euclidean"), abs=1e-14
        )
    pd.testing.assert_frame_equal(
        scores, pd.read_csv(results / "silhouette_scores.csv"), check_exact=False
    )
    selected = json.loads((results / "q1_selection.json").read_text())
    winner = scores.loc[scores.silhouette.idxmax()]
    assert (selected["K"], selected["H"]) == (winner.K, winner.H)
    assert selected["best_run_id"] == winner.best_run_id
    assert selected["silhouette"] == winner.silhouette
    assert selected["dataset_sha256"] == history["metadata"]["dataset_sha256"]
    assert (plots / "silhouette_heatmap.png").read_bytes().startswith(b"\x89PNG\r\n\x1a\n")
    assert (results / "q1_full_history.pkl").read_bytes() == original_checkpoint
    assert "Reduced experiment" in caplog.text

@pytest.mark.parametrize("corruption, message", [
    ("incomplete", "complete Q1 history"),
    ("labels", "Invalid winner labels"),
    ("scale", "standardization"),
    ("objective", "non-finite objectives"),
    ("version", "pandas_version"),
    ("format", "Unsupported Q1 checkpoint"),
])
def test_invalid_history_fails_before_output(experiment, corruption, message):
    data, results, plots = experiment
    history = read_history(results)
    if corruption == "incomplete":
        history["runs"].pop()
    elif corruption == "labels":
        for run in history["runs"]:
            run["labels"] = np.zeros(24, dtype=int)
    elif corruption == "scale":
        history["scaler_scale"][0] = 0
    elif corruption == "objective":
        history["runs"][0]["objective"] = np.nan
    elif corruption == "version":
        history["metadata"]["pandas_version"] = "different"
    else:
        history["metadata"]["format_version"] = 999
    write_history(results, history)
    with pytest.raises(ValueError, match=message):
        silhouette.analyze_silhouette(data, results, plots)
    assert not (results / "silhouette_scores.csv").exists()
    assert not plots.exists()

def test_nonconverged_winner_warns(experiment, caplog):
    data, results, plots = experiment
    history = read_history(results)

    write_history(results, history)
    scores = silhouette.analyze_silhouette(data, results, plots)
    assert not scores.best_converged.any()
    assert "did not converge" in caplog.text

import pandas as pd
import pytest

from src.analysis.statistical_analysis import (
    CLASSIFIER_NAMES,
    DATASET_NAMES,
    compute_confidence_intervals,
    run_statistical_analysis,
    validate_results,
)
from src.utils.statistical_tests import perform_friedman_nemenyi


def _make_cv_results() -> pd.DataFrame:
    rows = []
    for dataset_index, dataset in enumerate(DATASET_NAMES):
        for classifier_index, classifier in enumerate(CLASSIFIER_NAMES):
            for repetition in range(1, 31):
                for fold_in_rep in range(1, 11):
                    fold_id = (repetition - 1) * 10 + fold_in_rep - 1
                    score = (
                        0.4
                        + 0.08 * classifier_index
                        + 0.01 * dataset_index
                        + 0.0001 * fold_id
                    )
                    rows.append(
                        {
                            "fold_id": fold_id,
                            "repetition": repetition,
                            "fold_in_rep": fold_in_rep,
                            "dataset": dataset,
                            "classifier": classifier,
                            "error": 1 - score,
                            "precision": score,
                            "recall": score,
                            "f1": score,
                        }
                    )
    return pd.DataFrame(rows)


def test_validate_results_accepts_complete_30_by_10_results():
    df_results = _make_cv_results()

    validate_results(df_results)


def test_validate_results_rejects_missing_fold():
    df_results = _make_cv_results().iloc[1:]

    with pytest.raises(ValueError, match="Expected 3000"):
        validate_results(df_results)


def test_confidence_intervals_summarize_30_repetition_means():
    df_results = _make_cv_results()

    confidence_intervals = compute_confidence_intervals(df_results)

    assert len(confidence_intervals) == 40
    assert set(confidence_intervals["n_repetitions"]) == {30}
    assert (confidence_intervals["ci_lower"] <= confidence_intervals["mean"]).all()
    assert (confidence_intervals["mean"] <= confidence_intervals["ci_upper"]).all()


def test_friedman_nemenyi_returns_statistics_and_posthoc():
    classifier_count = len(CLASSIFIER_NAMES)
    scores = pd.DataFrame(
        {
            classifier: [
                0.2 + 0.15 * index + 0.001 * block
                for block in range(30)
            ]
            for index, classifier in enumerate(CLASSIFIER_NAMES)
        }
    )

    result = perform_friedman_nemenyi(scores, "synthetic metric")

    assert classifier_count == 5
    assert result.statistic > 0
    assert result.p_value < 0.05
    assert result.nemenyi_p_values is not None
    assert result.nemenyi_p_values.shape == (5, 5)


def test_friedman_rejects_undefined_identical_results():
    scores = pd.DataFrame(
        {classifier: [0.5] * 10 for classifier in CLASSIFIER_NAMES}
    )

    with pytest.raises(ValueError, match="undefined"):
        perform_friedman_nemenyi(scores, "identical metric")
